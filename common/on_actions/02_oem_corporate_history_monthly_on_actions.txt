# OEM-owned monthly corporate-history callbacks kept outside upstream event on-actions.
# This prevents upstream performance and civil-war cleanup changes from repeatedly
# conflicting with Corporate History during fork synchronization.

	on_monthly_FIN = {
		effect = {
			FIN_corporate_history_monthly_outcomes = yes
		}
	}

	on_monthly_TAI = {
		effect = {
			TAI_corporate_history_monthly_outcomes = yes
		}
	}

	on_monthly_JAP = {
		effect = {
			JAP_corporate_history_monthly_outcomes = yes
		}
	}

	on_monthly_CAN = {
		effect = {
			CAN_corporate_history_monthly_outcomes = yes
		}
	}
}
