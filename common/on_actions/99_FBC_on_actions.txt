on_actions = {
	on_startup = {
		effect = {
			random_country = { FBC_initialize_scenario = yes }
		}
	}

	# on_monthly fires for every country; the flag runs the pulse once a month.
	on_monthly = {
		effect = {
			if = {
				limit = {
					FBC_scenario_enabled = yes
					NOT = { has_global_flag = FBC_monthly_pulse_done }
				}
				set_global_flag = { flag = FBC_monthly_pulse_done value = 1 days = 27 }
				FBC_monthly_pulse = yes
			}
		}
	}
}
