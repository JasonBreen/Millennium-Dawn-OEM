# Corporate History game-rule gates. These wrap has_game_rule directly instead of
# cached GAME_RULE_* flags: the startup driver runs from 00_on_actions.txt, which
# sorts before 999_game_rules_on_actions.txt, so no cached flag exists yet there.
# Game rules are fixed at game setup, so the result never changes mid-game.

corporate_history_full_enabled = {
	NOT = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
	NOT = { has_game_rule = { rule = rule_corporate_history option = disabled } }
}

corporate_history_outcomes_only_enabled = {
	has_game_rule = { rule = rule_corporate_history option = outcomes_only }
}

# Single-statement gate for "Full or Outcomes Only". Never OR the two gates above:
# corporate_history_full_enabled has two children, so inlining it into an OR turns
# its implicit AND into an OR and the result is true even under Disabled.
corporate_history_enabled = {
	NOT = { has_game_rule = { rule = rule_corporate_history option = disabled } }
}
