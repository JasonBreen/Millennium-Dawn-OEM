# The eight major powers the Illuminati work inside.
DEUSEX_is_illuminati_target = {
	OR = {
		tag = USA
		tag = CHI
		tag = GER
		tag = ENG
		tag = FRA
		tag = SOV
		tag = JAP
		tag = RAJ
	}
}

# Host scope. Once the agenda reaches its third stage the Act is the Illuminati's, so a target that adopts it gives
# them 20 influence.
DEUSEX_restoration_act_serves_illuminati = {
	check_variable = { global.DEUSEX_illum_agenda > 2 }
	DEUSEX_is_illuminati_target = yes
	NOT = { has_country_flag = DEUSEX_inner_circle }
}

# Member scope. The agenda's next stage is within reach.
DEUSEX_agenda_ready = {
	OR = {
		AND = {
			check_variable = { global.DEUSEX_illum_agenda = 0 }
			check_variable = { global.DEUSEX_illum_media > 39 }
		}
		AND = {
			check_variable = { global.DEUSEX_illum_agenda = 1 }
			check_variable = { global.DEUSEX_illum_capital > 49 }
			date > 2025.1.1
		}
		AND = {
			check_variable = { global.DEUSEX_illum_agenda = 2 }
			check_variable = { global.DEUSEX_illum_security > 49 }
			has_global_flag = DEUSEX_aug_incident_done
		}
		AND = {
			check_variable = { global.DEUSEX_illum_agenda = 3 }
			check_variable = { global.DEUSEX_illum_capital > 39 }
			check_variable = { global.DEUSEX_illum_media > 39 }
			check_variable = { global.DEUSEX_illum_security > 39 }
		}
	}
}
