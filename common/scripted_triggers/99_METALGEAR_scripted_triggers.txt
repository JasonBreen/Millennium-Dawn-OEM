# The rule is unavailable while load-time triggers resolve; startup records it in a flag.
METALGEAR_scenario_enabled = {
	has_global_flag = GLOBAL_METALGEAR_scenario_enabled
}

# Country scope. The Patriots act through the United States' controller, human or AI, and no one else.
METALGEAR_is_controller = {
	METALGEAR_scenario_enabled = yes
	has_global_flag = GLOBAL_METALGEAR_network_ready
	original_tag = USA
}

# Temp mg_i set. Node mg_i is free for a new audit or repair.
METALGEAR_node_idle = {
	check_variable = { global.METALGEAR_node_audit^mg_i = 0 }
	check_variable = { global.METALGEAR_node_repair^mg_i = 0 }
}

# The network has a free commitment.
METALGEAR_has_capacity = {
	custom_trigger_tooltip = {
		tooltip = METALGEAR_has_capacity_tt
		check_variable = { global.METALGEAR_capacity > 0 }
	}
}
