# The rule is unavailable while load-time triggers resolve; startup records it in a flag.
FBC_scenario_enabled = {
	has_global_flag = GLOBAL_FBC_scenario_enabled
}

# USA scope. The station the polar case opened on still exists, is still the case country's, and still holds a rig.
FBC_polar_case_rig_stands = {
	check_variable = { FBC_polar_station_id > 0 }
	check_variable = { global.antarctica_station_exists^FBC_polar_station_id > 0 }
	check_variable = { global.antarctica_station_controller^FBC_polar_station_id = FBC_polar_country }
	OR = {
		check_variable = { global.antarctica_station_output_fuel_usage^FBC_polar_station_id < 1 }
		check_variable = { global.antarctica_station_output_storage^FBC_polar_station_id > 0 }
	}
	check_variable = { var = global.antarctica_station_output_power^FBC_polar_station_id value = 0 compare = greater_than_or_equals }
	var:FBC_polar_country = { exists = yes }
	var:FBC_polar_country = { NOT = { has_country_flag = FBC_polar_rig_down } }
	OR = {
		AND = {
			check_variable = { global.antarctica_station_module_slot_6^FBC_polar_station_id = 12 }
			NOT = { check_variable = { global.antarctica_station_module_slot_6_blizzard_damaged^FBC_polar_station_id > 0 } }
		}
		AND = {
			check_variable = { global.antarctica_station_module_slot_7^FBC_polar_station_id = 12 }
			NOT = { check_variable = { global.antarctica_station_module_slot_7_blizzard_damaged^FBC_polar_station_id > 0 } }
		}
		AND = {
			check_variable = { global.antarctica_station_module_slot_8^FBC_polar_station_id = 12 }
			NOT = { check_variable = { global.antarctica_station_module_slot_8_blizzard_damaged^FBC_polar_station_id > 0 } }
		}
		AND = {
			check_variable = { global.antarctica_station_module_slot_9^FBC_polar_station_id = 12 }
			NOT = { check_variable = { global.antarctica_station_module_slot_9_blizzard_damaged^FBC_polar_station_id > 0 } }
		}
	}
}
