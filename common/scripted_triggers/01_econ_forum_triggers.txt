# Economic forums (Issue #4802). See 01_econ_forum_effects.txt.
# Program tracks: 0 AI and technology, 1 energy, 2 finance, 3 defense industry, 4 infrastructure and trade, 5 development.

# Country scope, temp ef_i set. Members the forum always invites.
econ_forum_is_core_member = {
	OR = {
		AND = {
			check_variable = { ef_i = 1 }
			OR = {
				original_tag = POL
				original_tag = CZE
				original_tag = HUN
				original_tag = SLO
			}
		}
		AND = {
			check_variable = { ef_i = 2 }
			OR = {
				original_tag = BLR
				original_tag = KAZ
				original_tag = ARM
				original_tag = KYR
				original_tag = TAJ
			}
		}
		AND = {
			check_variable = { ef_i = 3 }
			has_country_flag = asian_nation_flag
		}
		AND = {
			check_variable = { ef_i = 4 }
			check_variable = { gdp_per_capita < 20 }
			is_in_array = { global.PR_regional_or_greater_powers = THIS.id }
		}
		AND = {
			check_variable = { ef_i = 5 }
			has_country_flag = sub_saharan_nation_flag
		}
		AND = {
			check_variable = { ef_i = 6 }
			econ_forum_is_arctic_nation = yes
		}
		AND = {
			check_variable = { ef_i = 7 }
			has_idea = NATO_member
		}
	}
}

econ_forum_is_arctic_nation = {
	OR = {
				original_tag = NRY
				original_tag = DEN
				original_tag = ICE
				original_tag = FIN
				original_tag = SWE
				original_tag = CAN
				original_tag = SOV
				original_tag = USA
	}
}

# Country scope, temp ef_t set. Whether track ef_t matters to this government.
econ_forum_interested_in_track = {
	OR = {
		AND = {
			check_variable = { ef_t = 0 }
			check_variable = { gdp_per_capita > 19.999 }
		}
		AND = {
			check_variable = { ef_t = 1 }
			OR = {
				check_variable = { energy_balance < 0 }
				check_variable = { oil_exports > 1 }
			}
		}
		AND = {
			check_variable = { ef_t = 2 }
			OR = {
				check_variable = { gdp_total > 499 }
				check_variable = { gdp_per_capita > 49.999 }
			}
		}
		AND = {
			check_variable = { ef_t = 3 }
			OR = {
				has_war = yes
				has_idea = defence_04
				has_idea = defence_05
				has_idea = defence_06
				has_idea = defence_07
				has_idea = defence_08
				has_idea = defence_09
			}
		}
		AND = {
			check_variable = { ef_t = 4 }
			check_variable = { gdp_per_capita < 20 }
		}
		AND = {
			check_variable = { ef_t = 5 }
			check_variable = { gdp_per_capita < 7 }
		}
	}
}

# Invitee scope, PREV is the host. True while the invitee has no personal invitation, reply or refusal this cycle.
econ_forum_can_be_invited = {
	OR = {
		check_variable = { econ_forum_status^num < 8 }
		AND = {
			check_variable = { econ_forum_status^PREV.econ_forum_hosted > -1 }
			check_variable = { econ_forum_status^PREV.econ_forum_hosted < 2 }
		}
	}
}

# Temp ef_after_last set: months from the forum's last summit to the next one. Summits stay at least
# 11 months apart, so a reschedule cannot fit a second summit into the same year.
econ_forum_summit_spacing_ok = {
	check_variable = { ef_after_last > 10 }
}

# Country scope, temp ef_host set. NATO and EU governments stay away from a host under Western sanctions.
econ_forum_boycotts_host = {
	OR = {
		has_idea = NATO_member
		has_idea = EU_member
	}
	var:ef_host = { has_country_flag = SOV_western_sanctions }
}

# Country scope, temp ef_host set. A full BRICS member hosting invites every other full member.
econ_forum_brics_standing_guest = {
	has_idea = RAJ_BRICS
	var:ef_host = { has_idea = RAJ_BRICS }
}

# Country scope, temp ef_host set. Full and associate BRICS members favour each other's forums.
econ_forum_brics_partner = {
	OR = {
		has_idea = RAJ_BRICS
		has_idea = RAJ_BRICS_associate
	}
	var:ef_host = {
		OR = {
			has_idea = RAJ_BRICS
			has_idea = RAJ_BRICS_associate
		}
	}
}

# Country scope, temp ef_host set. A host that pivoted to the East courts BRICS members and Asian nations.
econ_forum_pivot_east_guest = {
	var:ef_host = { has_country_flag = econ_forum_pivot_east }
	OR = {
		has_idea = RAJ_BRICS
		has_idea = RAJ_BRICS_associate
		has_country_flag = asian_nation_flag
	}
}

# Forums tab row of forum v, in the viewing country's scope. Someone else runs it.
econ_forum_guest_row = {
	check_variable = { global.econ_forum_host^v > 0 }
	NOT = { check_variable = { global.econ_forum_host^v = ROOT.id } }
}

econ_forum_can_sponsor = {
	econ_forum_guest_row = yes
	var:global.econ_forum_host^v = { NOT = { has_war_with = ROOT } }
	check_variable = { global.econ_forum_preparing^v = 1 }
	check_variable = { global.econ_forum_sponsors^v < 25 }
	NOT = { check_variable = { econ_forum_sponsor_cycle^v = global.econ_forum_cycle^v } }
	check_variable = { econ_forum_boycott^v = 0 }
	has_political_power > 9
	check_variable = { treasury > 1.49 }
}

econ_forum_can_bid = {
	econ_forum_guest_row = yes
	check_variable = { global.econ_forum_preparing^v = 1 }
	check_variable = { global.econ_forum_speakers^v < 3 }
	check_variable = { econ_forum_status^v > 2 }
	NOT = { check_variable = { econ_forum_bid_cycle^v = global.econ_forum_cycle^v } }
	has_political_power > 24
}

econ_forum_can_boycott = {
	econ_forum_guest_row = yes
	check_variable = { econ_forum_boycott^v = 0 }
	has_political_power > 24
}
