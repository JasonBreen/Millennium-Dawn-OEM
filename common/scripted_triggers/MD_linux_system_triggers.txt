# Linux-system rule gates read the game rule directly because startup runs before
# cached game-rule flags are available.

linux_system_full_enabled = {
	NOT = { has_game_rule = { rule = rule_linux_ecosystem option = outcomes_only } }
	NOT = { has_game_rule = { rule = rule_linux_ecosystem option = off } }
}

linux_system_outcomes_only_enabled = {
	has_game_rule = { rule = rule_linux_ecosystem option = outcomes_only }
}

linux_system_enabled = {
	OR = {
		linux_system_full_enabled = yes
		linux_system_outcomes_only_enabled = yes
	}
}

linux_system_historical_upstream_route = {
	OR = {
		original_tag = ENG
		original_tag = GER
		original_tag = BRA
	}
}

	OR = {
		original_tag = FRA

	}

linux_system_is_participant = {
	is_in_array = { global.linux_system_participants = THIS }
}
