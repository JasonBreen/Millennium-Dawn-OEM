defined_text = {
	name = STALKER_current_emission_risk

	text = {
		trigger = { has_state_flag = STALKER_first_emission_pending }
		localization_key = STALKER_first_emission_outlook
	}

	text = {
		trigger = {
			NOT = { has_state_flag = STALKER_first_emission_pending }
			has_state_flag = STALKER_emission_cooldown
		}
		localization_key = STALKER_emission_cooldown_outlook
	}

	text = {
		trigger = {
			NOT = { has_state_flag = STALKER_first_emission_pending }
			NOT = { has_state_flag = STALKER_emission_cooldown }
			set_temp_variable = {
				var = STALKER_emission_risk_display
				value = {
					value = STALKER_zone_activity
					add = STALKER_zone_exploitation
					subtract = STALKER_zone_containment
					multiply = 0.2
					clamp = { min = 2 max = 20 }
				}
			}
		}
		localization_key = STALKER_current_emission_risk_value
	}
}
