# Corporate-history framework. Docs: docs/src/content/resources/corporate-history-framework.md
#
# Chains: IBM, Sun/Microsoft, HP, Apple, NVIDIA, Dell, TI, Micron, Motorola (USA),
# Lenovo and Huawei (CHI), Sony (JAP), ATI/AMD and Matrox (CAN), Nokia (FIN), TSMC and Foxconn (TAI),
# Industrial Sovereignty (POL), Siemens (GER), Ericsson (SWE), and BlackBerry (CAN/USA).
# The rule_corporate_history game rule gates chains ONLY at dispatcher level: the
# startup driver below, the TAG_corporate_trigger_year_* effects
# (00_corporate_history_dispatch_effects.txt), the monthly outcome drivers below,
# and the USA_ibm_monthly_crisis_checks call site - never inside individual events.
#
# Start-date invariant: every milestone day offset assumes scheduling from January 1
# (startup at the 2000.1.1 bookmark) or from the first monthly tick of the year
# (yearly dispatch). Start-year schedulers therefore stay behind a per-year window
# (NOT = { has_start_date < Y.1.1 } + has_start_date < Y.1.2, see
# USA_apple_schedule_current_year_events); non-January starts get passed milestones
# from the reconstruct effects and deliberately do not queue the start year's rest.

# Effect: corporate_history_apply_delta
# Purpose: Applies a signed delta to a corporate state value and clamps it to the
#          framework band. The 0..10 bounds live only here.
# Parameters:
# - corp_delta: The signed delta to add before clamping
# - corp_value: The state value, read and written back as a temp variable (optional here only when chained from corporate_history_clamp_value; direct callers must set it)
corporate_history_apply_delta = {
	add_to_temp_variable = { corp_value = corp_delta }
	clamp_temp_variable = { var = corp_value min = 0 max = 10 }
}

# Effect: corporate_history_clamp_value
# Purpose: Clamps a corporate state value to the framework band without changing it.
# Parameters:
# - corp_value: The state value, read and written back as a temp variable
corporate_history_clamp_value = {
	set_temp_variable = { corp_delta = 0 }
	corporate_history_apply_delta = yes
}

# Effect: corporate_history_on_startup
# Purpose: Startup dispatcher for the corporate-history chains; sole caller is
#          MD_event_on_startup_events, which runs in the ABK init scope - every
#          action below must stay inside an explicit TAG = { } block.
corporate_history_on_startup = {
	if = {
		limit = { corporate_history_full_enabled = yes }
		if = {
			limit = { country_exists = USA }
			USA = {
				USA_microsoft_reconstruct_history = yes
				USA_apple_reconstruct_history = yes
				USA_hp_reconstruct_history = yes
				USA_google_reconstruct_history = yes
				USA_blackberry_reconstruct_history = yes
				USA_apple_schedule_current_year_events = yes
				USA_nvidia_schedule_current_year_events = yes
				USA_google_schedule_current_year_events = yes
				USA_blackberry_schedule_current_year_events = yes
				country_event = USA_micron_events.90
				country_event = USA_ti_events.90
				country_event = USA_motorola_events.90
				USA_ti_schedule_current_year_events = yes
				USA_micron_schedule_current_year_events = yes
				USA_motorola_schedule_current_year_events = yes
				USA_ibm_schedule_pc_decision = yes
				USA_ibm_schedule_x86_decision = yes
				country_event = { id = USA_e3_events.90 days = 1 }
				country_event = { id = USA_ibm_events.90 days = 1 }
				country_event = { id = USA_nvidia_events.90 days = 4 }
				USA_dell_schedule_current_year_events = yes
				country_event = { id = USA_dell_events.90 days = 5 }
				set_temp_variable = { aig_schedule_mode = 0 }
				USA_aig_schedule_current_year_events = yes
				country_event = { id = USA_aig_events.90 days = 6 }
				country_event = { id = USA_xbox_events.90 days = 1 }
				USA_corporate_systems_update_economic_bridge = yes
			}
		}

		if = {
			limit = { country_exists = CHI }
			CHI = {
				CHI_lenovo_schedule_current_year_events = yes
				country_event = { id = CHI_lenovo_events.90 days = 2 }
				CHI_huawei_reconstruct_history = yes
				CHI_huawei_schedule_current_year_events = yes
				country_event = { id = CHI_huawei_events.90 days = 3 }
			}
		}

		if = {
			limit = { country_exists = CAN }
			CAN = {
				CAN_ati_reconstruct_history = yes
				CAN_blackberry_reconstruct_history = yes
				CAN_ati_schedule_current_year_events = yes
				CAN_blackberry_schedule_current_year_events = yes
			}
		}
		# Reconstructs synchronously before scheduling so the current-year scheduler
		# sees every elapsed milestone already resolved. At the 2000.1.1 bookmark every
		# dated branch is still in the future, so this is a no-op there.
		if = {
			limit = { country_exists = POL }
			POL = {
				POL_industrial_sovereignty_initialize_state = yes
				POL_industrial_sovereignty_reconstruct_history = yes
				POL_industrial_sovereignty_schedule_current_year_events = yes
			}
		}

		if = {
			limit = {
				country_exists = USA
				NOT = { has_start_date < 2000.1.1 }
				has_start_date < 2000.1.2
			}
			USA = { country_event = { id = USA_hp_events.1 days = 153 } }
		}

		if = {
			limit = {
				country_exists = CAN
				date > 2000.3.15
			}
			CAN = { country_event = { id = CAN_matrox_events.90 days = 1 } }
		}
		else_if = {
			limit = { country_exists = CAN }
			CAN = { country_event = { id = CAN_matrox_events.1 days = 75 } }
		}

		if = {
			limit = { country_exists = FIN }
			FIN = {
				FIN_nokia_reconstruct_history = yes
				FIN_nokia_schedule_current_year_events = yes
				country_event = { id = FIN_nokia_events.90 days = 1 }
			}
		}

		if = {
			limit = { country_exists = GER }
			GER = {
				GER_siemens_reconstruct_history = yes
				GER_siemens_schedule_current_year_events = yes
			}
		}

		if = {
			limit = { country_exists = TAI }
			TAI = {
				TAI_tsmc_schedule_current_year_events = yes
				country_event = TAI_foxconn_events.90
				TAI_foxconn_schedule_current_year_events = yes
				country_event = { id = TAI_tsmc_events.90 days = 3 }
			}
		}

		if = {
			limit = { country_exists = SWE }
			SWE = {
				SWE_ericsson_schedule_current_year_events = yes
				country_event = { id = SWE_ericsson_events.90 days = 1 }
			}
		}

		if = {
			limit = {
				country_exists = JAP
				date > 2000.3.3
			}
			JAP = { country_event = { id = JAP_sony_events.90 days = 2 } }
		}
		else_if = {
			limit = { country_exists = JAP }
			JAP = { country_event = { id = JAP_sony_events.1 days = 63 } }
		}

		# Nintendo reconstructs synchronously before scheduling so the current-year
		# scheduler sees every prior-year predecessor already resolved.
		if = {
			limit = { country_exists = JAP }
			JAP = {
				JAP_nintendo_initialize_state = yes
				JAP_nintendo_reconstruct_history = yes
				set_temp_variable = { nin_schedule_mode = 0 }
				JAP_nintendo_schedule_current_year_events = yes
				country_event = { id = JAP_nintendo_events.90 days = 2 }
			}
		}

		if = {
			limit = {
				country_exists = FRA
				FRA = {
					original_tag = FRA
					NOT = { has_country_flag = collapsed_nation }
				}
			}
			FRA = {
				FRA_corporate_systems_initialize_state = yes
				FRA_corporate_systems_sync_nokia_response = yes
				FRA_corporate_systems_schedule_current_year_events = yes
				country_event = { id = FRA_corporate_systems_events.90 days = 1 }
			}
		}
	}
	else_if = {
		limit = { corporate_history_outcomes_only_enabled = yes }
		if = {
			limit = { country_exists = USA }
			USA = {
				USA_microsoft_reconstruct_history = yes
				USA_apple_reconstruct_history = yes
				USA_ibm_reconstruct_history = yes
				USA_nvidia_reconstruct_history = yes
				USA_google_reconstruct_history = yes
				USA_oracle_reconstruct_history = yes
				USA_hp_reconstruct_history = yes
				USA_e3_reconstruct_history = yes
				USA_dell_reconstruct_history = yes
				USA_micron_reconstruct_history = yes
				USA_ti_reconstruct_history = yes
				USA_motorola_reconstruct_history = yes
				USA_aig_reconstruct_history = yes
				USA_xbox_reconstruct_history = yes
				USA_blackberry_reconstruct_history = yes
				USA_corporate_systems_update_economic_bridge = yes
			}
		}
		if = {
			limit = { country_exists = JAP }
			JAP = {
				JAP_sony_reconstruct_history = yes
				JAP_nintendo_reconstruct_history = yes
			}
		}
		if = {
			limit = { country_exists = CAN }
			CAN = {
				CAN_ati_reconstruct_history = yes
				CAN_matrox_reconstruct_history = yes
				CAN_blackberry_reconstruct_history = yes
			}
		}
		if = {
			limit = { country_exists = POL }
			POL = {
				POL_industrial_sovereignty_reconstruct_history = yes
			}
		}
		if = {
			limit = { country_exists = FIN }
			FIN = { FIN_nokia_reconstruct_history = yes }
		}
		if = {
			limit = { country_exists = GER }
			GER = { GER_siemens_reconstruct_history = yes }
		}
		if = {
			limit = { country_exists = SWE }
			SWE = { SWE_ericsson_reconstruct_history = yes }
		}
		if = {
			limit = { country_exists = TAI }
			TAI = {
				TAI_tsmc_reconstruct_history = yes
				TAI_foxconn_reconstruct_history = yes
			}
		}
		if = {
			limit = { country_exists = CHI }
			CHI = {

				CHI_huawei_reconstruct_history = yes
			}
		}

			limit = {

					original_tag = FRA
					NOT = { has_country_flag = collapsed_nation }
				}
			}
			FRA = { FRA_corporate_systems_reconstruct_history = yes }
		}
	}
}
