# Corporate-history yearly dispatch.
# One effect per country per year, called from the matching trigger_year_*_events
# block; gated on the rule_corporate_history game rule at dispatcher level.
# Framework: 00_corporate_history_effects.txt; docs: docs/src/content/resources/corporate-history-framework.md

# Persistent catch-up driver for the Google extension chain (events 16-20) and the
# sole scheduling owner for them. The monthly driver preserves canonical dates in
# a normal campaign and queues one short-delay event per tick after a late start.
USA_google_extension_catchup = {
	if = {
		limit = { NOT = { has_country_flag = USA_google_extension_pending } }
		if = {
			limit = {
				date > 2009.12.31
				NOT = { has_country_flag = USA_google_event_16_resolved }
				NOT = { OR = {
					has_country_flag = USA_google_cables_allied_mesh
					has_country_flag = USA_google_cables_state_backed_corridors
					has_country_flag = USA_google_cables_lease_capacity
				} }
			}
			set_country_flag = { flag = USA_google_extension_pending value = 1 days = 400 }
			if = {
				limit = { date < 2010.2.1 }
				country_event = { id = USA_google_events.16 days = 89 }
			}
			else = {
				country_event = { id = USA_google_events.16 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2015.12.31
				OR = {
					has_country_flag = USA_google_cables_allied_mesh
					has_country_flag = USA_google_cables_state_backed_corridors
					has_country_flag = USA_google_cables_lease_capacity
				}
				NOT = { has_country_flag = USA_google_event_17_resolved }
			}
			set_country_flag = { flag = USA_google_extension_pending value = 1 days = 400 }
			if = {
				limit = { date < 2016.2.1 }
				country_event = { id = USA_google_events.17 days = 347 }
			}
			else = {
				country_event = { id = USA_google_events.17 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2020.12.31
				OR = {
					has_country_flag = USA_google_waymo_full_investment
					has_country_flag = USA_google_waymo_partnered_development
					has_country_flag = USA_google_waymo_minimal_investment
				}
				NOT = { has_country_flag = USA_google_event_18_resolved }
			}
			set_country_flag = { flag = USA_google_extension_pending value = 1 days = 400 }
			if = {
				limit = { date < 2021.2.1 }
				country_event = { id = USA_google_events.18 days = 137 }
			}
			else = {
				country_event = { id = USA_google_events.18 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2022.12.31
				OR = {
					has_country_flag = USA_google_vertex_ai_platform
					has_country_flag = USA_google_vertex_partnered_ai
					has_country_flag = USA_google_vertex_open_ecosystem
				}
				NOT = { has_country_flag = USA_google_event_19_resolved }
			}
			set_country_flag = { flag = USA_google_extension_pending value = 1 days = 400 }
			if = {
				limit = { date < 2023.2.1 }
				country_event = { id = USA_google_events.19 days = 79 }
			}
			else = {
				country_event = { id = USA_google_events.19 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2024.12.31
				OR = {
					has_country_flag = USA_google_bard_full_integration
					has_country_flag = USA_google_bard_partnered_ecosystem
					has_country_flag = USA_google_bard_open_competition
				}
				OR = {
					has_country_flag = USA_google_antitrust_settle_distribution
					has_country_flag = USA_google_antitrust_fight_distribution
					has_country_flag = USA_google_antitrust_structural_split
				}
				NOT = { has_country_flag = USA_google_event_20_resolved }
				NOT = { OR = {
					has_country_flag = USA_google_antitrust_settlement
					has_country_flag = USA_google_antitrust_litigation
					has_country_flag = USA_google_antitrust_structural_separation
				} }
			}
			set_country_flag = { flag = USA_google_extension_pending value = 1 days = 400 }
			if = {
				limit = { date < 2025.2.1 }
				country_event = { id = USA_google_events.20 days = 110 }
			}
			else = {
				country_event = { id = USA_google_events.20 days = 5 }
			}
		}
	}
}

# Persistent catch-up driver for the whole Oracle chain (events 1-12) and its
# sole scheduling owner. Sun ownership selects whether events 3-4 run; all paths
# reconverge at event 5. Same-year steps use delays from the next monthly tick.
USA_oracle_chain_catchup = {
	if = {
		limit = { NOT = { has_country_flag = USA_oracle_chain_pending } }
		if = {
			limit = {
				date > 2002.12.31
				NOT = { has_country_flag = USA_oracle_event_1_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2003.2.1 }
				country_event = { id = USA_oracle_events.1 days = 159 }
			}
			else = {
				country_event = { id = USA_oracle_events.1 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2007.12.31
				has_country_flag = USA_oracle_event_1_resolved
				NOT = { has_country_flag = USA_oracle_event_2_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2008.2.1 }
				country_event = { id = USA_oracle_events.2 days = 119 }
			}
			else = {
				country_event = { id = USA_oracle_events.2 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2009.12.31
				has_country_flag = USA_oracle_event_2_resolved
				has_country_flag = USA_sun_oracle_acquisition
				NOT = { has_country_flag = USA_oracle_event_3_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2010.2.1 }
				country_event = { id = USA_oracle_events.3 days = 26 }
			}
			else = {
				country_event = { id = USA_oracle_events.3 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2009.12.31
				has_country_flag = USA_oracle_event_3_resolved
				NOT = { has_country_flag = USA_oracle_event_4_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2010.3.1 }

		else_if = {
			limit = {
				date > 2011.12.31

				OR = {
					has_country_flag = USA_oracle_event_4_resolved
					has_country_flag = USA_sun_independent_restructuring
					has_country_flag = USA_sun_foundation_stewardship
					has_country_flag = USA_sun_microsoft_full_integration

				NOT = { has_country_flag = USA_oracle_event_5_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2012.2.1 }
				country_event = { id = USA_oracle_events.5 days = 144 }
			}
			else = {
				country_event = { id = USA_oracle_events.5 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2018.12.31
				has_country_flag = USA_oracle_event_5_resolved
				NOT = { has_country_flag = USA_oracle_event_6_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2019.2.1 }
				country_event = { id = USA_oracle_events.6 days = 287 }
			}
			else = {
				country_event = { id = USA_oracle_events.6 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2019.12.31
				has_country_flag = USA_oracle_event_6_resolved
				NOT = { has_country_flag = USA_oracle_event_7_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2020.2.1 }
				country_event = { id = USA_oracle_events.7 days = 260 }
			}
			else = {
				country_event = { id = USA_oracle_events.7 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2019.12.31
				has_country_flag = USA_oracle_event_7_resolved
				NOT = { has_country_flag = USA_oracle_event_8_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2020.11.1 }
				country_event = { id = USA_oracle_events.8 days = 26 }
			}

				country_event = { id = USA_oracle_events.8 days = 5 }

				date > 2019.12.31

				date > 2020.12.31
				has_country_flag = USA_oracle_event_10_resolved
				NOT = { has_country_flag = USA_oracle_event_11_resolved }

			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }

				limit = { date < 2021.2.1 }

			limit = {
				date > 2020.12.31
				has_country_flag = USA_oracle_event_11_resolved
				NOT = { has_country_flag = USA_oracle_event_9_resolved }
			}
			set_country_flag = { flag = USA_oracle_chain_pending value = 1 days = 400 }
			if = {
				limit = { date < 2021.7.1 }
				country_event = { id = USA_oracle_events.9 days = 202 }
			}
			else = {
				country_event = { id = USA_oracle_events.9 days = 5 }
			}
		}
		else_if = {
			limit = {
				date > 2023.12.31
				has_country_flag = USA_oracle_event_9_resolved
				NOT = { has_country_flag = USA_oracle_event_12_resolved }
			}

	}
