USA_spacex_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_spacex_state_initialized } }
		# Baselines chosen to reflect an early-2000s private launcher with limited cadence and contracts
		set_variable = { USA_spacex_launch_cadence_reliability = 1 }
		set_variable = { USA_spacex_reusability_depth = 1 }
		set_variable = { USA_spacex_government_partnership = 1 }
		set_variable = { USA_spacex_geostrategic_access = 1 }
		set_country_flag = USA_spacex_state_initialized
	}
	USA_spacex_clamp_state = yes
}

USA_spacex_clamp_state = {
	set_temp_variable = { corp_value = USA_spacex_launch_cadence_reliability }
	corporate_history_clamp_value = yes
	set_variable = { USA_spacex_launch_cadence_reliability = corp_value }
	set_temp_variable = { corp_value = USA_spacex_reusability_depth }
	corporate_history_clamp_value = yes
	set_variable = { USA_spacex_reusability_depth = corp_value }
	set_temp_variable = { corp_value = USA_spacex_government_partnership }
	corporate_history_clamp_value = yes
	set_variable = { USA_spacex_government_partnership = corp_value }
	set_temp_variable = { corp_value = USA_spacex_commercial_market_power }

USA_spacex_clear_capstone_outcome = {
	remove_ideas = {

}

USA_spacex_resolve_capstone = {
	USA_spacex_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_spacex_capstone_resolved } }
		# NVIDIA-style: resolve only by route flags; fallback is mixed-provider
		if = { limit = { has_country_flag = USA_spacex_route_cadence } USA_spacex_apply_cadence_leadership = yes }
		else_if = { limit = { has_country_flag = USA_spacex_route_starlink } USA_spacex_apply_starlink_infrastructure = yes }
		else_if = { limit = { has_country_flag = USA_spacex_route_starship } USA_spacex_apply_heavy_lift_transition = yes }
		else_if = { limit = { has_country_flag = USA_spacex_route_monopoly } USA_spacex_apply_commercial_monopoly_risk = yes }
		else_if = {
			limit = { OR = { has_country_flag = USA_spacex_route_mixed has_country_flag = USA_spacex_route_public_contract } }
			USA_spacex_apply_public_contract_backbone = yes
		}
		else = { USA_spacex_apply_public_contract_backbone = yes }
	}
}

# Hidden Outcomes Only reconstruction — chronological, reward-free

USA_spacex_reconstruct_outcomes_history = {
	USA_spacex_initialize_state = yes

	# 2002-03-14: Company founded
	if = {
		limit = { date > 2002.3.14 NOT = { has_country_flag = USA_spacex_founded } }

	}
	# 2006-08-01: COTS anchor (CRS follows in 2008 in narrative)
	if = {

		USA_spacex_record_cots_crs_awarded = yes

	# 2006-03-24: Falcon 1 first launch attempt (development pivot)
	if = {

		USA_spacex_record_falcon1_development = yes

	# 2008-09-28: Falcon 1 success
	if = {
		limit = { date > 2008.9.28 NOT = { has_country_flag = USA_spacex_falcon1_success } }
		USA_spacex_record_falcon1_success = yes
	}
	# 2010-06-04: Falcon 9 first flight
	if = {
		limit = { date > 2010.6.4 NOT = { has_country_flag = USA_spacex_falcon9_firstflight } }
		USA_spacex_record_falcon9_firstflight = yes
	}

	if = {
		limit = { date > 2012.5.25 NOT = { has_country_flag = USA_spacex_dragon_cargo_operational } }
		USA_spacex_record_cargo_to_iss = yes
	}
	# 2015-12-22: First stage landing
	if = {
		limit = { date > 2015.12.22 NOT = { has_country_flag = USA_spacex_first_stage_landing } }
		USA_spacex_record_first_landing = yes
	}
	# 2017-03-30: First reflight
	if = {
		limit = {
			date > 2017.3.30
			has_country_flag = USA_spacex_first_stage_landing
			NOT = { has_country_flag = USA_spacex_first_reflight }
		}
		USA_spacex_record_first_reflight = yes
	}
	# 2018-02-06: Falcon Heavy demo
	if = {
		limit = { date > 2018.2.6 NOT = { has_country_flag = USA_spacex_falcon_heavy_demo } }
		USA_spacex_record_falcon_heavy_demo = yes
	}
	# 2019-05-23: First large Starlink launch
	if = {
		limit = { date > 2019.5.23 NOT = { has_country_flag = USA_spacex_starlink_initial } }
		USA_spacex_record_starlink_initial = yes
	}
	# 2020-05-30: Crew Dragon Demo-2 (commercial crew)
	if = {
		limit = { date > 2020.5.30 NOT = { has_country_flag = USA_spacex_commercial_crew_operational } }
		USA_spacex_record_commercial_crew = yes
	}
	# 2021-04-16: HLS Option A
	if = {
		limit = { date > 2021.4.16 NOT = { has_country_flag = USA_spacex_hls_award } }
		USA_spacex_record_hls_award = yes
	}
	# 2021-08-01: NSSL dual-source operational composite
	if = {
		limit = { date > 2021.8.1 NOT = { has_country_flag = USA_spacex_nssl_dual_source } }
		USA_spacex_record_nssl_dual_source = yes
	}
	# 2023-04-20: Starship integrated flight test
	if = {
		limit = { date > 2023.4.20 NOT = { has_country_flag = USA_spacex_starship_integrated_test } }
		USA_spacex_record_starship_integrated = yes
	}
	# 2023-04-20+: Iterative licensing posture
	if = {
		limit = { date > 2023.4.20 NOT = { has_country_flag = USA_spacex_iterative_licensing } }
		USA_spacex_record_starship_iterative_licensing = yes
	}
	# 2024-06-06: Starship progress (IFT-4)
	if = {
		limit = { date > 2024.6.6 NOT = { has_country_flag = USA_spacex_starship_progress_2024 } }
		USA_spacex_record_starship_progress_2024 = yes
	}

	USA_spacex_clamp_state = yes

	# Historical-A route choice applied silently at the route window when none is set
	if = {
		limit = {
			date > 2026.8.31
			NOT = { OR = {
				has_country_flag = USA_spacex_route_cadence
				has_country_flag = USA_spacex_route_starlink
				has_country_flag = USA_spacex_route_starship
				has_country_flag = USA_spacex_route_monopoly
				has_country_flag = USA_spacex_route_mixed
				has_country_flag = USA_spacex_route_public_contract
			} }
		}
		USA_spacex_record_route_cadence = yes
	}

	if = {
		limit = { date > 2026.9.1 NOT = { has_country_flag = USA_spacex_capstone_resolved } }
		USA_spacex_resolve_capstone = yes
	}
}

		# 2002
		if = {

			USA_spacex_dispatch_2002 = yes
		}

			USA_spacex_dispatch_2006 = yes
		}

		if = {
			limit = { NOT = { has_start_date < 2008.1.1 } has_start_date < 2008.1.2 NOT = { has_country_flag = USA_spacex_falcon1_success } }
			USA_spacex_dispatch_2008 = yes
		}
