			NOT = {
				AND = {
					OR = {
						has_country_flag = GER_siemens_integrated_group
						has_country_flag = GER_siemens_digital_industry
					}
					OR = {
						has_country_flag = GER_siemens_networks_joint_venture
						has_country_flag = GER_siemens_independent_automation
					}
					OR = {
						has_country_flag = GER_siemens_sale_reinvestment
						has_country_flag = GER_siemens_network_partnerships
					}
					OR = {
						has_country_flag = GER_siemens_healthineers_listed
						has_country_flag = GER_siemens_healthineers_integrated
					}
					OR = {
						has_country_flag = GER_siemens_energy_spun_off
						has_country_flag = GER_siemens_energy_integrated
					}
				}
			}
		}

			limit = { corporate_history_outcomes_only_enabled = yes }

			if = {

		}
	}
