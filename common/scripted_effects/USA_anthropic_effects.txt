		limit = { NOT = { has_country_flag = USA_anthropic_state_initialized } }
		set_variable = { USA_anthropic_governance_strength = 8 }
		set_variable = { USA_anthropic_frontier_capability = 2 }
		set_variable = { USA_anthropic_commercial_reach = 0 }
		set_variable = { USA_anthropic_safety_discipline = 8 }
		set_variable = { USA_anthropic_cloud_independence = 4 }

USA_anthropic_resolve_capstone = {
	USA_anthropic_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_anthropic_capstone_resolved } }
		if = {
			limit = { check_variable = { USA_anthropic_capstone_choice = 1 } }
			USA_anthropic_apply_safety_governed_frontier = yes
		}
		else_if = {
			limit = { check_variable = { USA_anthropic_capstone_choice = 2 } }
			USA_anthropic_apply_cloud_aligned_scale = yes
		}
		else_if = {
			limit = { check_variable = { USA_anthropic_capstone_choice = 3 } }
			USA_anthropic_apply_multi_cloud_independence = yes
		}
		else_if = {
			limit = { check_variable = { USA_anthropic_capstone_choice = 4 } }
			USA_anthropic_apply_cautious_research_laboratory = yes
		}
		else = { USA_anthropic_apply_safety_governed_frontier = yes }
	}
}
