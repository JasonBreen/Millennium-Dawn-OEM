USA_xai_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_xai_state_initialized } }
		set_variable = { USA_xai_frontier_capability = 1 }
		set_variable = { USA_xai_compute_scale = 3 }
		set_variable = { USA_xai_talent = 4 }
		set_variable = { USA_xai_deployment = 1 }
		set_variable = { USA_xai_public_standing = 5 }
		set_country_flag = USA_xai_state_initialized
	}
	USA_xai_clamp_state = yes
}

USA_xai_clamp_state = {
	clamp_variable = { var = USA_xai_frontier_capability min = 0 max = 10 }
	clamp_variable = { var = USA_xai_compute_scale min = 0 max = 10 }
	clamp_variable = { var = USA_xai_talent min = 0 max = 10 }
	clamp_variable = { var = USA_xai_deployment min = 0 max = 10 }
	clamp_variable = { var = USA_xai_public_standing min = 0 max = 10 }
}

USA_xai_apply_colossus_route = {
	USA_xai_initialize_state = yes
	add_to_variable = { USA_xai_compute_scale = 3 }
	add_to_variable = { USA_xai_frontier_capability = 2 }
	add_to_variable = { USA_xai_deployment = 1 }
	add_to_variable = { USA_xai_public_standing = -1 }
