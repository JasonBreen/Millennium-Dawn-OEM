TOP_cache_game_rule = {
	set_variable = { global.TOP_rule_mode = 0 }
	if = {
		limit = { has_game_rule = { rule = TOP_game_rule option = TOP_limited_sandbox_option } }
		set_variable = { global.TOP_rule_mode = 1 }
	}
	else_if = {
		limit = { has_game_rule = { rule = TOP_game_rule option = TOP_full_sandbox_option } }
		set_variable = { global.TOP_rule_mode = 2 }
	}
}

TOP_enable_diagnostics = {
	set_global_flag = TOP_diagnostics_enabled
}

TOP_initialize_global = {
	if = {
		limit = { TOP_enabled = yes check_variable = { global.TOP_status^num = 0 } }
		set_variable = { global.TOP_clock = 0 }
		TOP_setup_registry = yes
	}
}

TOP_country_initialize = {
	if = {
		limit = { TOP_enabled = yes check_variable = { TOP_known^num = 0 } }
		TOP_resize_country_arrays = yes
	}
}

TOP_global_weekly = {
	TOP_cache_game_rule = yes
	if = {
		limit = { TOP_enabled = yes }
		TOP_initialize_global = yes
		add_to_variable = { global.TOP_clock = 7 }
	}
	if = {
		limit = { has_global_flag = TOP_diagnostics_enabled }
		set_temp_variable = { TOP_probe_registry = global.TOP_status^num }
		log = "TOP_PROBE mode=[?global.TOP_rule_mode] clock=[?global.TOP_clock] registry=[?TOP_probe_registry] country_ticks=[?global.TOP_diag_country_ticks]"
		set_variable = { global.TOP_diag_country_ticks = 0 }
	}
}

TOP_country_tick = {
	if = {
		limit = { TOP_enabled = yes }
		TOP_country_initialize = yes
		if = {
			limit = { has_global_flag = TOP_diagnostics_enabled }
			add_to_variable = { global.TOP_diag_country_ticks = 1 }
		}
	}
}
