# Planning owns preferences and savings targets, never stage progress or external resource balances.
ai_race_ai_register_country = {
	if = {
		limit = { NOT = { is_in_array = { global.ai_race_ai_countries = THIS } } }
		add_to_array = { global.ai_race_ai_countries = THIS }
	}
	if = {
		limit = { NOT = { has_variable = ai_race_ai_target_stage } }
		set_variable = { ai_race_ai_target_stage = 0 }
	}
}

ai_race_ai_discover_candidates = {
	if = {
		limit = {
			ai_race_full_mode = yes
			date > 2012.12.31
			NOT = { check_variable = { global.ai_race_ai_discovery_year = global.year } }
		}
		set_variable = { global.ai_race_ai_discovery_year = global.year }
		for_each_scope_loop = {
			array = global.countries
			if = {
				limit = {
					is_ai = yes
					exists = yes
					NOT = { has_country_flag = disabled_economic_system }
					NOT = { has_country_flag = collapsed_nation }
					check_variable = { gdp_total > 0 }
				}
				set_temp_variable = {
					ai_race_ai_candidate_civs = {
						value = 0
						every_collection = {
							named_collection = controlled_states_collection
							add = non_damaged_building_level@industrial_complex
						}
					}
				}
				if = {
					limit = {
						OR = {
							ai_race_ai_candidate_eligible = yes
							check_variable = { ai_race_stage > 0 }
						}
					}
					ai_race_ai_register_country = yes
				}
			}
		}
	}
}

ai_race_ai_refresh_plans = {
	if = {
		limit = { ai_race_full_mode = yes date > 2012.12.31 }
		for_each_scope_loop = {
			array = global.ai_race_all_initialized
			if = {
				limit = { is_ai = yes check_variable = { ai_race_stage > 0 } }
				ai_race_ai_register_country = yes
			}
		}
		for_each_scope_loop = {
			array = global.ai_race_ai_countries
			ai_race_ai_update_plan = yes
		}
	}
}

ai_race_ai_load_target = {
	set_temp_variable = { ai_race_requirement_stage = ai_race_ai_target_stage }
	ai_race_load_stage_requirements = yes
	set_variable = { ai_race_ai_target_cost = ai_race_requirement_cost }
	set_variable = { ai_race_ai_civs_required = ai_race_requirement_civs }
	set_variable = { ai_race_ai_network_required = ai_race_requirement_network }
	set_variable = { ai_race_ai_power_required = { value = ai_race_requirement_power multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_microchips_required = ai_race_requirement_microchips }
	set_variable = { ai_race_ai_composites_required = ai_race_requirement_composites }
	set_variable = { ai_race_ai_oil_required = ai_race_requirement_oil }
	ai_race_ai_refresh_readiness = yes
}

ai_race_ai_choose_target = {
	ai_race_get_month_token = yes
	set_temp_variable = { ai_race_ai_old_target = ai_race_ai_target_stage }
	set_temp_variable = { ai_race_ai_old_recovery = ai_race_ai_recovery }
	set_variable = { ai_race_ai_target_stage = ai_race_stage }
	set_variable = { ai_race_ai_recovery = 0 }
	set_variable = { ai_race_ai_planning_purchase = 0 }
	ai_race_ai_load_target = yes
	set_variable = { ai_race_ai_paid_power_required = ai_race_ai_power_required }
	if = {
		limit = {
			check_variable = { ai_race_stage > 0 }
			OR = {
				check_variable = { ai_race_ai_readiness < 1 }
				AND = {
					check_variable = { ai_race_ai_old_target = ai_race_stage }
					check_variable = { ai_race_ai_old_recovery = 1 }
					OR = {
						AND = {
							check_variable = { ai_race_ai_power_latched = 1 }
							check_variable = { ai_race_ai_power_ratio < 1.1 }
						}
						AND = {
							check_variable = { ai_race_ai_microchips_latched = 1 }
							check_variable = { ai_race_ai_microchips_ratio < 1.1 }
						}
						AND = {
							check_variable = { ai_race_ai_composites_latched = 1 }
							check_variable = { ai_race_ai_composites_ratio < 1.1 }
						}
					}
				}
			}
		}
		set_variable = { ai_race_ai_recovery = 1 }
	}
	else_if = {
		limit = { check_variable = { ai_race_stage < 4 } }
		set_variable = { ai_race_ai_earliest_entry_month = 24193 }
		if = {
			limit = { check_variable = { ai_race_stage = 1 } }
			set_variable = { ai_race_ai_earliest_entry_month = 24229 }
		}
		else_if = {
			limit = { check_variable = { ai_race_stage = 2 } }
			set_variable = { ai_race_ai_earliest_entry_month = 24289 }
		}
		else_if = {
			limit = { check_variable = { ai_race_stage = 3 } }
			set_variable = { ai_race_ai_earliest_entry_month = 24301 }
		}
		set_temp_variable = {
			ai_race_ai_finish_month = {
				value = ai_race_requirement_months
				subtract = ai_race_stage_months
				max = 0
				add = ai_race_current_month
			}
		}
		set_variable = {
			ai_race_ai_earliest_entry_month = {
				value = ai_race_ai_earliest_entry_month
				max = ai_race_ai_finish_month
			}
		}
		set_temp_variable = { ai_race_ai_lookahead_month = { value = ai_race_current_month add = 36 } }
		if = {
			limit = {
				OR = {
					check_variable = { ai_race_stage = 0 }
					check_variable = { var = ai_race_ai_lookahead_month value = ai_race_ai_earliest_entry_month compare = greater_than_or_equals }
				}
			}
			add_to_variable = { ai_race_ai_target_stage = 1 }
			set_variable = { ai_race_ai_planning_purchase = 1 }
		}
	}
	if = {
		limit = { NOT = { check_variable = { ai_race_ai_target_stage = ai_race_ai_old_target } } }
		set_variable = { ai_race_ai_power_latched = 0 }
		set_variable = { ai_race_ai_microchips_latched = 0 }
		set_variable = { ai_race_ai_composites_latched = 0 }
	}
	ai_race_ai_load_target = yes
}

ai_race_ai_update_plan = {
	if = {
		limit = { ai_race_ai_country_operable = yes }
		ai_race_refresh_planning_sample = yes
		ai_race_ai_choose_target = yes
		ai_race_ai_cache_construction = yes
		ai_race_ai_choose_construction = yes
		ai_race_ai_choose_research = yes
		ai_race_ai_calculate_savings = yes
	}
	else = { ai_race_ai_reset_preferences = yes }
}

# This refresh never buys a stage, collects an installment, or changes sovereign debt.
ai_race_ai_refresh_savings = {
	if = {
		limit = {
			has_variable = ai_race_ai_target_stage
			ai_race_ai_country_operable = yes
		}
		ai_race_refresh_planning_sample = yes
		ai_race_ai_load_target = yes
		ai_race_ai_calculate_savings = yes
	}
	else = { set_variable = { ai_race_ai_savings_target = 0 } }
}

ai_race_ai_calculate_savings = {
	set_variable = { ai_race_ai_savings_target = 0 }
	set_variable = { ai_race_ai_projected_payment = 0 }
	if = {
		limit = {
			ai_race_ai_expansion_permitted = yes
			check_variable = { ai_race_ai_planning_purchase = 1 }
			check_variable = { ai_race_ai_target_stage > ai_race_stage }
			check_variable = { var = ai_race_ai_readiness value = 0.75 compare = greater_than_or_equals }
		}
		set_variable = {
			ai_race_ai_savings_target = {
				value = display_expense
				max = 0
				multiply = 52
				add = ai_race_ai_target_cost
			}
		}
		if = {
			limit = { check_variable = { ai_race_ai_readiness < 1 } }
			set_temp_variable = {
				ai_race_ai_possible_payment = {
					value = 1
					subtract = ai_race_ai_readiness
					multiply = 0.4
					add = 0.45
					multiply = gdp_total
					divide = 520
				}
			}
			set_temp_variable = { ai_race_ai_projected_balance = { value = treasury_rate subtract = ai_race_ai_possible_payment } }
			set_temp_variable = { ai_race_ai_debt_limit = { value = gdp_total multiply = 0.6 } }
			if = {
				limit = {
					check_variable = { debt < ai_race_ai_debt_limit }
					check_variable = { var = ai_race_ai_projected_balance value = 0 compare = greater_than_or_equals }
				}
				set_variable = { ai_race_ai_projected_payment = ai_race_ai_possible_payment }
				add_to_variable = { ai_race_ai_savings_target = { value = ai_race_ai_possible_payment multiply = 52 } }
			}
		}
	}
}

				add = building_level@nuclear_facility
			}
		}
	}

	set_variable = { ai_race_ai_slots_network = 0 }

		if = {
			limit = { free_building_slots = { building = fossil_powerplant size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_fossil = 1 }
		}
		if = {
			limit = { free_building_slots = { building = renewable_energy_infra size > 0 } }

		}
		if = {
			limit = { free_building_slots = { building = nuclear_reactor size > 0 } }

			limit = { free_building_slots = { building = microchip_plant size > 0 } }

		}
		if = {
			limit = { free_building_slots = { building = nuclear_facility size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_facility = 1 }
		}
	}
	set_variable = { ai_race_ai_network_workers = { value = 12 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_fossil_workers = { value = 7.5 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_renewable_workers = { value = 22.5 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_nuclear_workers = { value = 115 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_microchip_plant_power = { value = 1 add = modifier@energy_use_modifier_microchip_plant max = 0 multiply = 0.75 multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_composite_plant_power = { value = 1 add = modifier@energy_use_modifier_composite_plant max = 0 multiply = 0.8 multiply = ai_race_power_multiplier } }

	set_variable = { ai_race_ai_refinery_power = { value = 1 add = modifier@energy_use_modifier_synthetic_refinery max = 0 multiply = 0.2 multiply = ai_race_power_multiplier } }

	set_variable = { ai_race_ai_new_reactor_fuel = { value = 1 add = modifier@nuclear_fuel_consumption_modifier max = 0.0001 multiply = 40 } }
	set_variable = { ai_race_ai_reactor_fuel_reserve = { value = nuclear_fuel_consumption add = ai_race_ai_new_reactor_fuel multiply = 52 } }

	# Preparing one real supporting plant may require more power than the race's own demand.
	set_temp_variable = { ai_race_ai_added_plant_power = 0 }
	if = {
		limit = {
			check_variable = { ai_race_ai_civs_ratio < 1 }
			ai_race_ai_expansion_permitted = yes
			check_variable = { ai_race_ai_civs_total < ai_race_ai_civs_required }
			check_variable = { ai_race_ai_slots_civs > 0 }

			check_variable = { var = resource@chromium value = 2 compare = greater_than_or_equals }

		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_microchip_plant_power } }
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_composites_ratio < 1 }
			check_variable = { ai_race_ai_slots_composites > 0 }
			has_tech = composite_production_1
			can_staff_an_composite_plant = yes
			check_variable = { var = resource@rubber value = 1 compare = greater_than_or_equals }
			check_variable = { var = resource@chromium value = 1 compare = greater_than_or_equals }
			check_variable = { var = resource@oil value = 1 compare = greater_than_or_equals }
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_composite_plant_power } }
	}
	if = {
		limit = {
			ai_race_ai_rubber_feedstock_needed = yes
			ai_race_ai_refinery_base_capacity = yes
			has_tech = fuel_silos
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_refinery_power } }
	}
	set_variable = { ai_race_ai_power_build_required = { value = ai_race_ai_paid_power_required add = ai_race_ai_added_plant_power max = ai_race_ai_power_required } }
	set_variable = { ai_race_ai_power_build_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_power_build_required > 0 } }
		set_variable = { ai_race_ai_power_build_ratio = { value = ai_race_power_available divide = ai_race_ai_power_build_required clamp = { min = 0 max = 2 } } }
	}
	ai_race_ai_choose_power_source = yes
}

ai_race_ai_choose_power_source = {
	set_variable = { ai_race_ai_power_source = 0 }
	set_temp_variable = { ai_race_ai_best_power_score = 0 }
	if = {
		limit = { ai_race_ai_can_add_fossil_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@fossil_pp_energy_generation_modifier max = 0 multiply = 2 multiply = { value = 1 add = modifier@production_speed_fossil_powerplant_factor max = 0 } multiply = 100000 divide = 9500 } }
		set_temp_variable = { ai_race_ai_best_power_score = ai_race_ai_power_score }
		set_variable = { ai_race_ai_power_source = 1 }
	}
	if = {
		limit = { ai_race_ai_can_add_renewable_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@renewable_energy_gain_multiplier max = 0 multiply = 0.25 multiply = { value = 1 add = modifier@production_speed_renewable_energy_infra_factor max = 0 } multiply = 100000 divide = 50000 } }
		if = {
			limit = { check_variable = { ai_race_ai_power_score > ai_race_ai_best_power_score } }
			set_temp_variable = { ai_race_ai_best_power_score = ai_race_ai_power_score }
			set_variable = { ai_race_ai_power_source = 2 }
		}
	}
	if = {
		limit = { ai_race_ai_can_add_nuclear_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@nuclear_energy_generation_modifier max = 0 multiply = 4 multiply = { value = 1 add = modifier@production_speed_nuclear_reactor_factor max = 0 } multiply = 100000 divide = 50000 } }
		if = {
			limit = { check_variable = { ai_race_ai_power_score > ai_race_ai_best_power_score } }
			set_variable = { ai_race_ai_power_source = 3 }
		}
	}
}

		ai_race_ai_consider_ai_tech = { TECH = artificial_intelligence_8 CODE = 8 YEAR = 2019 }

		set_temp_variable = { ai_race_ai_best_research_shortfall = 0 }
		set_temp_variable = { ai_race_ai_research_shortfall = { value = 1 subtract = ai_race_ai_civs_ratio max = 0 } }

			limit = {
				ai_race_ai_expansion_permitted = yes
				check_variable = { ai_race_ai_research_shortfall > ai_race_ai_best_research_shortfall }
			}
			set_temp_variable = { ai_race_ai_best_research_shortfall = ai_race_ai_research_shortfall }
			set_variable = { ai_race_ai_support_need = 1 }
		}
		set_temp_variable = { ai_race_ai_research_shortfall = { value = 1 subtract = ai_race_ai_network_ratio max = 0 } }
		if = {
			limit = {
				ai_race_ai_expansion_permitted = yes
				check_variable = { ai_race_ai_research_shortfall > ai_race_ai_best_research_shortfall }
			}
			set_temp_variable = { ai_race_ai_best_research_shortfall = ai_race_ai_research_shortfall }
			set_variable = { ai_race_ai_support_need = 2 }
		}
		set_temp_variable = { ai_race_ai_research_shortfall = { value = 1 subtract = ai_race_ai_power_build_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_research_shortfall > ai_race_ai_best_research_shortfall } }
			set_temp_variable = { ai_race_ai_best_research_shortfall = ai_race_ai_research_shortfall }
			set_variable = { ai_race_ai_support_need = 3 }
		}
		set_temp_variable = { ai_race_ai_research_shortfall = { value = 1 subtract = ai_race_ai_microchips_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_research_shortfall > ai_race_ai_best_research_shortfall } }
			set_temp_variable = { ai_race_ai_best_research_shortfall = ai_race_ai_research_shortfall }
			set_variable = { ai_race_ai_support_need = 4 }
		}
		set_temp_variable = { ai_race_ai_research_shortfall = { value = 1 subtract = ai_race_ai_composites_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_research_shortfall > ai_race_ai_best_research_shortfall } }
			set_temp_variable = { ai_race_ai_best_research_shortfall = ai_race_ai_research_shortfall }
			set_variable = { ai_race_ai_support_need = 5 }
		}
		if = {
			limit = { check_variable = { ai_race_ai_support_need = 1 } }
			ai_race_ai_choose_construction_tech = yes
		}
		if = {
			limit = { check_variable = { ai_race_ai_support_need = 2 } }
			ai_race_ai_choose_network_tech = yes
		}
		if = {
			limit = { check_variable = { ai_race_ai_support_need = 4 } }
			ai_race_ai_choose_microchips_tech = yes
		}
		if = {
			limit = { check_variable = { ai_race_ai_support_need = 5 } }
			if = {
				limit = {
					ai_race_ai_rubber_feedstock_needed = yes
					OR = {
						check_variable = { synthetic_refinery_total > 0 }
						check_variable = { ai_race_ai_slots_refinery > 0 }
					}
				}
				ai_race_ai_choose_fossil_tech = yes
			}
			else = { ai_race_ai_choose_composites_tech = yes }
		}
		if = {
			limit = { check_variable = { ai_race_ai_support_need = 3 } }
			if = {
				limit = { ai_race_ai_fossil_fuel_support_needed = yes }
				ai_race_ai_choose_refining_tech = yes
			}
			else_if = {
				limit = { check_variable = { ai_race_ai_power_source = 1 } }
				ai_race_ai_choose_fossil_tech = yes
			}
			else_if = {
				limit = { check_variable = { ai_race_ai_power_source = 3 } }
				ai_race_ai_choose_nuclear_tech = yes
			}
			else = { ai_race_ai_choose_renewable_tech = yes }
			if = {
				limit = { check_variable = { ai_race_ai_support_tech = 0 } }
				ai_race_ai_choose_efficiency_tech = yes
			}
		}
	}
}
