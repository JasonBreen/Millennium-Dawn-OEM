# Nintendo corporate history. Five milestones set the route; the closing
# milestone resolves it into one of four futures for Japanese entertainment.

JAP_nintendo_apply_disciplined_hybrid = {
	if = {
		limit = { JAP_nintendo_outcome_settled = no }
		add_ideas = JAP_nintendo_disciplined_hybrid_platform
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
JAP_nintendo_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = JAP_electronics.310 days = 255 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = JAP_nintendo_dedicated_handheld
	}
}
