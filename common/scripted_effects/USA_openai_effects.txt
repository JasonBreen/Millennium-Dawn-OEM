USA_openai_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_openai_state_initialized } }
		set_variable = { USA_openai_mission_control = 8 }
		set_variable = { USA_openai_frontier_capability = 2 }
		set_variable = { USA_openai_deployment_reach = 0 }
		set_variable = { USA_openai_safety_governance = 6 }
		set_variable = { USA_openai_compute_independence = 3 }
		set_country_flag = USA_openai_state_initialized
	}
	USA_openai_clamp_state = yes
}

USA_openai_clamp_state = {

USA_openai_resolve_capstone = {
	USA_openai_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_openai_capstone_resolved } }
		if = {
			limit = { check_variable = { USA_openai_capstone_choice = 1 } }
			USA_openai_apply_nonprofit_controlled_pbc = yes
		}
		else_if = {
			limit = { check_variable = { USA_openai_capstone_choice = 2 } }
			USA_openai_apply_scaled_commercial_platform = yes
		}
		else_if = {
			limit = { check_variable = { USA_openai_capstone_choice = 3 } }
			USA_openai_apply_safety_governed_laboratory = yes
		}
		else_if = {
			limit = { check_variable = { USA_openai_capstone_choice = 4 } }
			USA_openai_apply_diversified_frontier_consortium = yes
		}
		else = { USA_openai_apply_nonprofit_controlled_pbc = yes }
	}
}
