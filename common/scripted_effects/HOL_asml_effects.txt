# ASML corporate-history: state, reconstruction, scheduling, dispatchers, and capstone
#
# Root prefix: HOL_asml
# Namespace: HOL_asml_events (events/HOL_asml_events.txt)
# Tag: HOL
#
# Conventions:
# - Variables are integers clamped 0..10 via corporate_history_clamp_value
# - All effects are idempotent and logged via event-option logs or dispatcher logs
# - Visible events are is_triggered_only and owned by the yearly dispatcher
#

HOL_asml_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = HOL_asml_state_initialized } }
		set_variable = { HOL_asml_lithography_leadership = 1 }
		set_variable = { HOL_asml_supply_chain_depth = 1 }
		set_variable = { HOL_asml_export_alignment = 1 }
		set_country_flag = HOL_asml_state_initialized
	}
	HOL_asml_clamp_state = yes
}

HOL_asml_clamp_state = {
	set_temp_variable = { corp_value = HOL_asml_lithography_leadership }
	corporate_history_clamp_value = yes
	set_variable = { HOL_asml_lithography_leadership = corp_value }
	set_temp_variable = { corp_value = HOL_asml_supply_chain_depth }
	corporate_history_clamp_value = yes
	set_variable = { HOL_asml_supply_chain_depth = corp_value }
	set_temp_variable = { corp_value = HOL_asml_export_alignment }
	corporate_history_clamp_value = yes
	set_variable = { HOL_asml_export_alignment = corp_value }
}

# Capstone outcomes (exactly one idea)

HOL_asml_clear_capstone_outcome = {
	remove_ideas = {
		HOL_asml_chokepoint_steward
		HOL_asml_triad_supply_sovereign
		HOL_asml_open_market_maximizer
		HOL_asml_dual_use_gatekeeper
		HOL_asml_balanced_atlantic_anchor
	}
	clr_country_flag = HOL_asml_capstone_resolved

HOL_asml_resolve_capstone = {
	HOL_asml_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = HOL_asml_capstone_resolved } }
		if = { limit = { has_country_flag = HOL_asml_route_chokepoint_steward } HOL_asml_apply_chokepoint_steward = yes }
		else_if = { limit = { has_country_flag = HOL_asml_route_triad_supply_sovereign } HOL_asml_apply_triad_supply_sovereign = yes }
		else_if = { limit = { has_country_flag = HOL_asml_route_open_market_maximizer } HOL_asml_apply_open_market_maximizer = yes }
		else_if = { limit = { has_country_flag = HOL_asml_route_dual_use_gatekeeper } HOL_asml_apply_dual_use_gatekeeper = yes }
		else = { HOL_asml_apply_balanced_anchor = yes }
	}
}

# Reconstruction

HOL_asml_reconstruct_history = {
	if = {
		limit = { corporate_history_outcomes_only_enabled = yes }
		HOL_asml_reconstruct_outcomes_history = yes
	}
	if = {
		limit = {
			date > 2026.9.1
			has_country_flag = HOL_asml_capstone_resolved
		}
		set_country_flag = HOL_asml_reconstruct_complete
	}
}

HOL_asml_reconstruct_outcomes_history = {
	HOL_asml_initialize_state = yes

	# 2003-06-01: Immersion bet committed
	if = {
		limit = { date > 2003.6.1 NOT = { has_country_flag = HOL_asml_immersion_bet_committed } }
		HOL_asml_record_immersion_bet = yes
	}
	# 2004-04-01: XT:1250i shipped
	if = {
		limit = { date > 2004.4.1 NOT = { has_country_flag = HOL_asml_xt1250i_shipped } }
		HOL_asml_record_xt1250i_delivery = yes
	}
	# 2006-10-01: Alpha Demo Tools broadened
	if = {
		limit = { date > 2006.10.1 NOT = { has_country_flag = HOL_asml_alpha_demo_tools_broadened } }
		HOL_asml_record_alpha_demo_tools = yes
	}
	# 2010-03-01: NXE:3100 first light
	if = {
		limit = { date > 2010.3.1 NOT = { has_country_flag = HOL_asml_nxe3100_first_light } }
		HOL_asml_record_nxe3100_first_light = yes
	}
	# 2012-07-01: CCIP expanded
	if = {
		limit = { date > 2012.7.1 NOT = { has_country_flag = HOL_asml_ccip_expanded } }
		HOL_asml_record_ccip = yes
	}
	# 2013-05-01: Cymer integrated
	if = {
		limit = { date > 2013.5.1 NOT = { has_country_flag = HOL_asml_cymer_integrated } }
		HOL_asml_record_cymer_acquisition = yes
	}
	# 2016-11-01: Zeiss SMT integration
	if = {
		limit = { date > 2016.11.1 NOT = { has_country_flag = HOL_asml_zeiss_smt_integration } }
		HOL_asml_record_zeiss_smt_stake = yes
	}
	# 2019-07-01: EUV HVM cadence
	if = {
		limit = { date > 2019.7.1 NOT = { has_country_flag = HOL_asml_euv_hvm_cadence } }
		HOL_asml_record_euv_hvm = yes
	}
	# 2019-12-01: EUV China license non-renewal
	if = {
		limit = { date > 2019.12.1 NOT = { has_country_flag = HOL_asml_euv_china_license_nonrenewal } }
		HOL_asml_record_euv_china_license_nonrenewal = yes
	}
	# 2023-03-01: Advanced DUV licensing posture tightened
	if = {
		limit = { date > 2023.3.1 NOT = { has_country_flag = HOL_asml_advanced_duv_controls } }
		HOL_asml_record_advanced_duv_licensing = yes
	}
	# 2024-06-01: High-NA EXE deployed to allies
	if = {
		limit = { date > 2024.6.1 NOT = { has_country_flag = HOL_asml_high_na_exe_deployed } }
		HOL_asml_record_high_na_exe = yes
	}

	HOL_asml_clamp_state = yes

	# Historical-A route choice applied silently at the route window when none is set
	if = {
		limit = {
			date > 2026.8.31
			NOT = { OR = {
				has_country_flag = HOL_asml_route_chokepoint_steward
				has_country_flag = HOL_asml_route_triad_supply_sovereign
				has_country_flag = HOL_asml_route_open_market_maximizer
				has_country_flag = HOL_asml_route_dual_use_gatekeeper
				has_country_flag = HOL_asml_route_balanced_anchor
			} }
		}
		HOL_asml_record_route_chokepoint_steward = yes
	}

	if = {
		limit = { date > 2026.9.1 NOT = { has_country_flag = HOL_asml_capstone_resolved } }
		HOL_asml_resolve_capstone = yes
	}
}
