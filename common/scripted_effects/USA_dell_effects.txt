USA_dell_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_dell_state_initialized } }

		set_variable = { USA_dell_direct_model_strength = 6 }
		set_variable = { USA_dell_enterprise_pivot = 1 }
		set_variable = { USA_dell_balance_sheet_leverage = 1 }
		set_variable = { USA_dell_founder_control = 5 }
		set_variable = { USA_dell_governance_legitimacy = 4 }
		set_country_flag = USA_dell_state_initialized
	}

	USA_dell_clamp_state = yes
}

USA_dell_clamp_state = {
	set_temp_variable = { corp_value = USA_dell_direct_model_strength }
	corporate_history_clamp_value = yes
	set_variable = { USA_dell_direct_model_strength = corp_value }
	set_temp_variable = { corp_value = USA_dell_enterprise_pivot }
	corporate_history_clamp_value = yes
	set_variable = { USA_dell_enterprise_pivot = corp_value }
	set_temp_variable = { corp_value = USA_dell_balance_sheet_leverage }
	corporate_history_clamp_value = yes
	set_variable = { USA_dell_balance_sheet_leverage = corp_value }
	set_temp_variable = { corp_value = USA_dell_founder_control }
	corporate_history_clamp_value = yes
	set_variable = { USA_dell_founder_control = corp_value }
	set_temp_variable = { corp_value = USA_dell_governance_legitimacy }
	corporate_history_clamp_value = yes
	set_variable = { USA_dell_governance_legitimacy = corp_value }
}

USA_dell_reconstruct_history = {

		limit = {
			date > 2004.7.16

				OR = {
					has_country_flag = USA_dell_rollins_direct_machine
					has_country_flag = USA_dell_founder_stays_ceo
				}
			}
		}
		set_country_flag = USA_dell_rollins_direct_machine
		add_to_variable = { USA_dell_direct_model_strength = 2 }

USA_dell_clear_capstone_outcome = {
	remove_ideas = {
		USA_dell_ai_infrastructure_backbone
		USA_dell_integrated_infrastructure_federation
		USA_dell_founder_private_empire
		USA_dell_disciplined_direct_manufacturer
		USA_dell_leveraged_restructurer
	}

	if = {
		limit = { has_country_flag = USA_dell_outcome_ai_backbone }
		clr_country_flag = USA_dell_outcome_ai_backbone
	}
	if = {
		limit = { has_country_flag = USA_dell_outcome_integrated_federation }
		clr_country_flag = USA_dell_outcome_integrated_federation
	}
	if = {
		limit = { has_country_flag = USA_dell_outcome_founder_empire }
		clr_country_flag = USA_dell_outcome_founder_empire
	}
	if = {
		limit = { has_country_flag = USA_dell_outcome_direct_manufacturer }
		clr_country_flag = USA_dell_outcome_direct_manufacturer
	}
	if = {
		limit = { has_country_flag = USA_dell_outcome_leveraged_restructurer }
		clr_country_flag = USA_dell_outcome_leveraged_restructurer
	}
	if = {
		limit = { has_country_flag = USA_dell_capstone_resolved }
		clr_country_flag = USA_dell_capstone_resolved
	}
}

USA_dell_resolve_capstone = {
	USA_dell_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = USA_dell_capstone_resolved } }

		if = {
			limit = {
				check_variable = { var = USA_dell_enterprise_pivot value = 8 compare = greater_than_or_equals }
				OR = {
					has_country_flag = USA_dell_ai_server_buildout
					has_country_flag = USA_dell_ai_full_stack
				}
			}
			USA_dell_apply_ai_backbone_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_dell_enterprise_pivot value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_dell_governance_legitimacy value = 6 compare = greater_than_or_equals }
			}
			USA_dell_apply_integrated_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_dell_founder_control value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_dell_balance_sheet_leverage value = 6 compare = greater_than_or_equals }
			}
			USA_dell_apply_founder_empire_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_dell_direct_model_strength value = 7 compare = greater_than_or_equals }
			}
			USA_dell_apply_direct_manufacturer_capstone = yes
		}
		else = { USA_dell_apply_leveraged_restructurer_capstone = yes }
	}
}
