# Reads top_agency_host in the initiating country.
top_reserve_agency_case = {
	if = {
		limit = { top_agency_uses_native_route = yes }
		set_variable = { top_agency_case_host = top_agency_host }
		set_variable = { top_agency_case_kind = TOP_operation_subject_kind }
		set_variable = { top_agency_case_sequence = TOP_operation_sequence }
		custom_effect_tooltip = top_agency_case_launch_tt
	}
}

# Reads top_agency_callback_host in the initiating country.
top_resolve_agency_case = {
	if = {
		limit = { top_agency_case_matches_host = yes }
		clear_variable = top_agency_case_sequence
		if = {
			limit = { check_variable = { TOP_operation_subject_kind = 1 } }
			set_temp_variable = { TOP_target = TOP_operation_subject_id }
			set_temp_variable = { TOP_arg_target = TOP_target }
			set_temp_variable = { TOP_method = TOP_case_method^TOP_target }
			set_temp_variable = { TOP_operation_state = TOP_case_state^TOP_target }
			set_temp_variable = { TOP_facility_kind = TOP_case_facility^TOP_target }
			set_temp_variable = { TOP_case_visit_target = TOP_target }
			TOP_load_case = yes
			if = {
				limit = {
					TOP_mission_binding_valid = yes
					NOT = { TOP_case_visit_waiting_for_arrival = yes }
				}
				TOP_resolve_timed_case = yes
			}
			else = {
				set_temp_variable = { TOP_arg_sequence = TOP_case_sequence^TOP_target }
				TOP_close_case = yes
			}
		}
		else_if = {
			limit = { check_variable = { TOP_operation_subject_kind = 2 } }
			set_temp_variable = { TOP_arg_group = TOP_operation_subject_id }
			if = { limit = { TOP_organization_mission_binding_valid = yes } TOP_resolve_organization_operation = yes }
			else = {
				set_temp_variable = { TOP_arg_sequence = TOP_org_case_sequence^TOP_arg_group }
				TOP_close_organization_case = yes
			}
		}
	}
}
