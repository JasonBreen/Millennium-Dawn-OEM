DEUSEX_monthly_germany_pulse = {
	if = {
		limit = { DEUSEX_scenario_enabled = yes country_exists = GER }
		GER = {
			if = {
				limit = {
					check_variable = { DEUSEX_exchange_phase = 1 }
					OR = {
						NOT = { country_exists = CHI }
						has_war_with = CHI
					}
				}
				set_variable = { DEUSEX_exchange_phase = 4 }
				country_event = { id = DEUSEX.23 days = 1 }
			}
		}
	}
}
