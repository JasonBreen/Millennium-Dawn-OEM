# Silent Hill. A Bureau case in Maine (state 764). The fog lives on the state; the case lives on USA with the Bureau.
# FBC_sh_phase: 0 dormant, 1 reported, 2 intervention running, 3 answered. FBC_sh_intervention: 1 a response team,
# 2 a quarantine, 3 a cover-up. The town answers the intervention when FBC_sh_answer_in reaches 0.
FBC_monthly_silenthill_pulse = {
	if = {
		limit = {
			FBC_scenario_enabled = yes
			country_exists = USA
			USA = { has_variable = FBC_containment_integrity }
			NOT = { has_global_flag = FBC_sh_ended }
		}
		if = {
			limit = { NOT = { has_global_flag = FBC_sh_ready } }
			set_global_flag = FBC_sh_ready
			764 = {
				set_variable = { FBC_sh_fog = 10 }
				FBC_update_silenthill_fog = yes
				add_dynamic_modifier = { modifier = FBC_silent_hill_fog }
			}
		}
		764 = {
			FBC_silenthill_fog_month = yes
			FBC_silenthill_town_month = yes
		}
		USA = {
			FBC_silenthill_case_month = yes
			FBC_silenthill_town_events_month = yes
			FBC_silenthill_ending_month = yes
		}
	}
}

# State 764 scope. The fog climbs by 1 a month; a response team in town holds it back, a cover-up lets it run.
# The town also feeds on its visitors: every 50 of them add 1 fog a month.
FBC_silenthill_fog_month = {
	set_temp_variable = { FBC_sh_fog_gain = 1 }
	if = {
		limit = { USA = { check_variable = { FBC_sh_committed > 0 } } }
		set_temp_variable = { FBC_sh_fog_gain = -1 }
	}
	else_if = {
		limit = {
			USA = {
				check_variable = { FBC_sh_phase = 2 }
				check_variable = { FBC_sh_intervention = 3 }
			}
		}
		set_temp_variable = { FBC_sh_fog_gain = 2 }
	}
	add_to_temp_variable = { FBC_sh_fog_gain = { value = FBC_sh_visitors multiply = 0.02 } }
	add_to_variable = { FBC_sh_fog = FBC_sh_fog_gain }
	FBC_update_silenthill_fog = yes
}

# State 764 scope. Call after every change to the fog.
FBC_update_silenthill_fog = {
	clamp_variable = { var = FBC_sh_fog min = 0 max = 100 }
	set_variable = { FBC_sh_supply_factor = { value = FBC_sh_fog multiply = -0.004 } }
	set_variable = { FBC_sh_building_factor = { value = FBC_sh_fog multiply = -0.003 } }
	set_variable = { FBC_sh_resource_factor = { value = FBC_sh_fog multiply = -0.002 } }
	set_variable = { FBC_sh_attrition_factor = { value = FBC_sh_fog multiply = 0.002 } }
}

# USA scope. Reports come at 25 fog; the town answers the intervention when its timer runs out. Both need the Bureau
# to hold the town. FBC.30 sets phase 1 when it arrives, so reports lost to a change of control are sent again.
FBC_silenthill_case_month = {
	if = {
		limit = {
			check_variable = { FBC_sh_phase = 0 }
			764 = {
				check_variable = { FBC_sh_fog > 24 }
				is_controlled_by = USA
			}
			NOT = { has_country_flag = FBC_sh_report_pending }
		}
		set_country_flag = { flag = FBC_sh_report_pending value = 1 days = 7 }
		country_event = { id = FBC.30 days = 1 }
	}
	else_if = {
		limit = {
			check_variable = { FBC_sh_phase = 2 }
			764 = { is_controlled_by = USA }
		}
		# A team intervention needs a team in town: recommit one after a lost town, or wait until one is free.
		if = {
			limit = {
				check_variable = { FBC_sh_intervention = 1 }
				check_variable = { FBC_sh_committed < 1 }
			}
			FBC_commit_silenthill_team = yes
		}
		if = {
			limit = {
				OR = {
					NOT = { check_variable = { FBC_sh_intervention = 1 } }
					check_variable = { FBC_sh_committed > 0 }
				}
			}
			add_to_variable = { FBC_sh_answer_in = -1 }
		}
		if = {
			limit = { check_variable = { FBC_sh_answer_in < 1 } }
			set_variable = { FBC_sh_phase = 3 }
			if = {
				limit = { check_variable = { FBC_sh_intervention = 1 } }
				random_list = {
					60 = { country_event = { id = FBC.31 days = 1 } }
					40 = { country_event = { id = FBC.32 days = 1 } }
				}
			}
			else_if = {
				limit = { check_variable = { FBC_sh_intervention = 2 } }
				country_event = { id = FBC.33 days = 1 }
			}
			else = { country_event = { id = FBC.34 days = 1 } }
		}
	}
	# Losing the town pauses the case and frees its team; the intervention stays on record.
	else_if = {
		limit = {
			check_variable = { FBC_sh_phase = 2 }
			check_variable = { FBC_sh_committed > 0 }
		}
		FBC_release_silenthill_team = yes
	}
}

# USA scope. Records the intervention and when the town answers it, in months.
FBC_silenthill_intervene = {
	set_variable = { FBC_sh_phase = 2 }
	set_variable = { FBC_sh_intervention = FBC_sh_choice }
	set_variable = { FBC_sh_answer_in = FBC_sh_delay }
}

FBC_commit_silenthill_team = {
	if = {
		limit = {
			check_variable = { FBC_response_capacity > 0 }
			check_variable = { FBC_sh_committed < 1 }
		}
		add_to_variable = { FBC_response_capacity = -1 }
		set_variable = { FBC_sh_committed = 1 }
	}
}

FBC_release_silenthill_team = {
	if = {
		limit = { check_variable = { FBC_sh_committed > 0 } }
		set_variable = { FBC_sh_committed = 0 }
		add_to_variable = { FBC_response_capacity = 1 }
		clamp_variable = { var = FBC_response_capacity min = 0 max = 2 }
	}
}

# USA scope, temp FBC_sh_fog_change set.
FBC_change_silenthill_fog = {
	764 = {
		add_to_variable = { FBC_sh_fog = FBC_sh_fog_change }
		FBC_update_silenthill_fog = yes
	}
}

# State 764 scope. Letters draw visitors while the fog is up, at a twentieth of the fog a month, half that under
# quarantine. Thick fog grows the Order by 1 a month plus 1 for every 50 visitors, half that while it is watched.
FBC_silenthill_town_month = {
	if = {
		limit = { check_variable = { FBC_sh_fog > 24 } }
		set_temp_variable = { FBC_sh_visitor_gain = { value = FBC_sh_fog multiply = 0.05 } }
		if = {
			limit = {
				USA = {
					check_variable = { FBC_sh_phase = 2 }
					check_variable = { FBC_sh_intervention = 2 }
				}
			}
			multiply_temp_variable = { FBC_sh_visitor_gain = 0.5 }
		}
		add_to_variable = { FBC_sh_visitors = FBC_sh_visitor_gain }
		clamp_variable = { var = FBC_sh_visitors min = 0 max = 100 }
	}
	if = {
		limit = { check_variable = { FBC_sh_fog > 49 } }
		set_temp_variable = { FBC_sh_order_gain = { value = FBC_sh_visitors multiply = 0.02 add = 1 } }
		if = {
			limit = { USA = { has_country_flag = FBC_sh_order_watched } }
			multiply_temp_variable = { FBC_sh_order_gain = 0.5 }
		}
		add_to_variable = { FBC_sh_order = FBC_sh_order_gain }
		clamp_variable = { var = FBC_sh_order min = 0 max = 100 }
	}
}

# USA scope. Once the case is on file, while the Bureau holds the town: the Siren at 40 fog, at 0.15% a month per
# point of fog; the letters at 20 visitors; the Order at 40, and its rite at 80 once the Bureau knows of it.
# Each event starts its own cooldown on arrival; a 7-day pending flag covers the queue, so one lost to a change of
# control goes out again.
FBC_silenthill_town_events_month = {
	if = {
		limit = {
			check_variable = { FBC_sh_phase > 0 }
			764 = { is_controlled_by = USA }
		}
		if = {
			limit = {
				764 = { check_variable = { FBC_sh_fog > 39 } }
				NOT = { has_country_flag = FBC_sh_siren_cooldown }
				NOT = { has_country_flag = FBC_sh_siren_pending }
			}
			764 = { set_temp_variable = { FBC_sh_siren_chance = { value = FBC_sh_fog multiply = 0.15 } } }
			random = {
				chance = FBC_sh_siren_chance
				set_country_flag = { flag = FBC_sh_siren_pending value = 1 days = 7 }
				country_event = { id = FBC.35 days = 1 }
			}
		}
		if = {
			limit = {
				764 = { check_variable = { FBC_sh_visitors > 19 } }
				NOT = { has_country_flag = FBC_sh_letters_cooldown }
				NOT = { has_country_flag = FBC_sh_letters_pending }
			}
			set_country_flag = { flag = FBC_sh_letters_pending value = 1 days = 7 }
			country_event = { id = FBC.36 days = 1 }
		}
		if = {
			limit = {
				NOT = { has_country_flag = FBC_sh_order_known }
				764 = { check_variable = { FBC_sh_order > 39 } }
				NOT = { has_country_flag = FBC_sh_order_pending }
			}
			set_country_flag = { flag = FBC_sh_order_pending value = 1 days = 7 }
			country_event = { id = FBC.37 days = 1 }
		}
		else_if = {
			limit = {
				has_country_flag = FBC_sh_order_known
				764 = { check_variable = { FBC_sh_order > 79 } }
				NOT = { has_country_flag = FBC_sh_rite_cooldown }
				NOT = { has_country_flag = FBC_sh_rite_pending }
			}
			set_country_flag = { flag = FBC_sh_rite_pending value = 1 days = 7 }
			country_event = { id = FBC.38 days = 1 }
		}
	}
}

# USA scope. One ending per game, while the case is on file and the Bureau holds the town. Rebirth after a finished
# rite at 90 fog. In Water at 80 exposure in fog of 50 or more, or containment under 20 in fog of 75 or more. Leave
# once the town has answered and fog, the Order and visitors have all fallen away. The joke endings need their own
# rule and come at 0.5% a month each; the dog needs a team in town.
FBC_silenthill_ending_month = {
	if = {
		limit = {
			check_variable = { FBC_sh_phase > 0 }
			764 = { is_controlled_by = USA }
			NOT = { has_country_flag = FBC_sh_ending_pending }
		}
		if = {
			limit = {
				has_global_flag = FBC_sh_rite_performed
				764 = { check_variable = { FBC_sh_fog > 89 } }
			}
			set_country_flag = { flag = FBC_sh_ending_pending value = 1 days = 7 }
			country_event = { id = FBC.41 days = 1 }
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						check_variable = { FBC_exposure > 79 }
						764 = { check_variable = { FBC_sh_fog > 49 } }
					}
					AND = {
						check_variable = { FBC_containment_integrity < 20 }
						764 = { check_variable = { FBC_sh_fog > 74 } }
					}
				}
			}
			set_country_flag = { flag = FBC_sh_ending_pending value = 1 days = 7 }
			country_event = { id = FBC.40 days = 1 }
		}
		else_if = {
			limit = {
				check_variable = { FBC_sh_phase = 3 }
				764 = {
					check_variable = { FBC_sh_fog < 15 }
					check_variable = { FBC_sh_order < 20 }
					check_variable = { FBC_sh_visitors < 10 }
				}
			}
			set_country_flag = { flag = FBC_sh_ending_pending value = 1 days = 7 }
			country_event = { id = FBC.39 days = 1 }
		}
		else_if = {
			limit = { has_game_rule = { rule = rule_fbc_silenthill_jokes option = ENABLED } }
			random_list = {
				198 = { }
				1 = {
					set_country_flag = { flag = FBC_sh_ending_pending value = 1 days = 7 }
					country_event = { id = FBC.42 days = 1 }
				}
				1 = {
					modifier = { factor = 0 check_variable = { FBC_sh_committed < 1 } }
					set_country_flag = { flag = FBC_sh_ending_pending value = 1 days = 7 }
					country_event = { id = FBC.43 days = 1 }
				}
			}
		}
	}
}

# USA scope, temp FBC_sh_ending_choice set: 1 Leave, 2 In Water, 3 Rebirth, 4 lights over the lake, 5 the dog.
# Closes the case for good and frees its team.
FBC_silenthill_end_case = {
	set_variable = { FBC_sh_ending = FBC_sh_ending_choice }
	set_global_flag = FBC_sh_ended
	FBC_release_silenthill_team = yes
}

# USA scope. The fog lifts and both of the town's modifiers go with it.
FBC_silenthill_lift_fog = {
	764 = {
		set_variable = { FBC_sh_fog = 0 }
		FBC_update_silenthill_fog = yes
		remove_dynamic_modifier = { modifier = FBC_silent_hill_fog }
		remove_dynamic_modifier = { modifier = FBC_silent_hill_otherworld }
	}
}
