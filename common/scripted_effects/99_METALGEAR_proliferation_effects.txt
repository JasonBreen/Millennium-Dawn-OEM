METALGEAR_pf_commit = {
	if = {
		limit = { tag = USA check_variable = { global.METALGEAR_capacity > 0 } check_variable = { METALGEAR_pf_committed < 1 } }
		add_to_variable = { global.METALGEAR_capacity = -1 }
		set_variable = { METALGEAR_pf_committed = 1 }
	}
}

METALGEAR_pf_release = {
	if = {
		limit = { check_variable = { METALGEAR_pf_committed > 0 } }
		set_variable = { METALGEAR_pf_committed = 0 }
		add_to_variable = { global.METALGEAR_capacity = 1 }
		clamp_variable = { var = global.METALGEAR_capacity min = 0 max = 2 }
	}
}

METALGEAR_pf_close_work = {
	METALGEAR_pf_release = yes
	clr_country_flag = METALGEAR_pf_work_wait
	clr_country_flag = METALGEAR_pf_notice_pending
}

METALGEAR_pf_interrupt = {
	if = {
		limit = { OR = { check_variable = { METALGEAR_pf_phase = 2 } check_variable = { METALGEAR_pf_phase = 3 } } }
		set_variable = { METALGEAR_pf_phase = 5 }
		METALGEAR_pf_close_work = yes
	}
}

METALGEAR_pf_usa_month = {
	if = {
		limit = { NOT = { country_exists = USA } }
		USA = { METALGEAR_pf_interrupt = yes }
	}
	else = {
		USA = {
			if = {
				limit = { date > 2005.1.1 OR = { check_variable = { METALGEAR_pf_phase = 0 } check_variable = { METALGEAR_pf_phase = 2 } check_variable = { METALGEAR_pf_phase = 3 } } }
				set_variable = { METALGEAR_pf_phase = 5 }
				METALGEAR_pf_close_work = yes
			}
			if = {
				limit = { check_variable = { METALGEAR_pf_phase = 2 } NOT = { has_country_flag = METALGEAR_pf_work_wait } }
				set_variable = { METALGEAR_pf_phase = 3 }
			}
			if = {
				limit = { check_variable = { METALGEAR_pf_phase = 3 } NOT = { has_country_flag = METALGEAR_pf_notice_pending } }
				set_country_flag = { flag = METALGEAR_pf_notice_pending value = 1 days = 35 }
				country_event = { id = METALGEAR.50 days = 1 }
			}
		}
	}
}

METALGEAR_pf_jap_exchange_month = {
	if = {
		limit = { JAP = { check_variable = { METALGEAR_pf_exchange = 1 } } }
		if = {
			limit = { OR = { NOT = { country_exists = JAP } NOT = { country_exists = USA } JAP = { has_war_with = USA } NOT = { USA = { check_variable = { METALGEAR_pf_inquiry = 2 } } } } }
			JAP = { set_variable = { METALGEAR_pf_exchange = 4 } }
		}
		else_if = {
			limit = { JAP = { NOT = { has_country_flag = METALGEAR_pf_exchange_notice } } }
			JAP = { set_country_flag = { flag = METALGEAR_pf_exchange_notice value = 1 days = 35 } country_event = { id = METALGEAR.54 days = 1 } }
		}
	}
}

METALGEAR_monthly_proliferation_pulse = {
	if = {
		limit = { METALGEAR_scenario_enabled = yes }
		METALGEAR_pf_usa_month = yes
		METALGEAR_pf_jap_month = yes
		METALGEAR_pf_sov_month = yes
		METALGEAR_pf_jap_exchange_month = yes
		METALGEAR_pf_sov_exchange_month = yes
		if = {
			limit = { NOT = { country_exists = USA } }
			USA = { METALGEAR_pf_release = yes METALGEAR_pf_close_inquiry = yes set_variable = { METALGEAR_pf_inquiry = 5 } }
		}
		else = {
			USA = {
				if = {
					limit = { date > 2005.3.1 check_variable = { METALGEAR_pf_inquiry = 0 } OR = { check_variable = { global.METALGEAR_shadow_moses_result > 0 } date > 2006.1.1 } }
					set_variable = { METALGEAR_pf_inquiry = 1 }
				}
				if = {
					limit = { 769 = { NOT = { is_controlled_by = USA } } }
					METALGEAR_pf_interrupt = yes
					if = { limit = { OR = { check_variable = { METALGEAR_pf_inquiry = 2 } check_variable = { METALGEAR_pf_inquiry = 3 } } } set_variable = { METALGEAR_pf_inquiry = 5 } METALGEAR_pf_close_inquiry = yes }
				}
				if = {
					limit = { check_variable = { METALGEAR_pf_inquiry = 2 } NOT = { has_country_flag = METALGEAR_pf_inquiry_wait } }
					set_variable = { METALGEAR_pf_inquiry = 3 }
				}
				if = {
					limit = { NOT = { has_country_flag = METALGEAR_pf_inquiry_notice } }
					if = { limit = { check_variable = { METALGEAR_pf_inquiry = 1 } } set_country_flag = { flag = METALGEAR_pf_inquiry_notice value = 1 days = 35 } country_event = { id = METALGEAR.53 days = 1 } }
					else_if = { limit = { check_variable = { METALGEAR_pf_inquiry = 3 } } set_country_flag = { flag = METALGEAR_pf_inquiry_notice value = 1 days = 35 } country_event = { id = METALGEAR.56 days = 1 } }
				}
			}
		}
	}
}
