ISR_oem_resolve_strategy = {
	if = {
		limit = {
			has_country_flag = ISR_oem_event_1_resolved
			has_country_flag = ISR_oem_event_2_resolved
			has_country_flag = ISR_oem_event_3_resolved
			has_country_flag = ISR_oem_event_4_resolved
			NOT = { check_variable = { ISR_oem_event_1_route = 0 } }
			NOT = { check_variable = { ISR_oem_event_2_route = 0 } }
			NOT = { check_variable = { ISR_oem_event_3_route = 0 } }
			NOT = { check_variable = { ISR_oem_event_4_route = 0 } }
		}
		remove_ideas = {
			ISR_oem_startup_nation_network
			ISR_oem_national_champions
			ISR_oem_strategic_consolidation
		}
		if = {
			limit = {
				check_variable = {
					var = ISR_oem_startup_network_score
					value = ISR_oem_national_champion_score
					compare = greater_than_or_equals
				}
				check_variable = {
					var = ISR_oem_startup_network_score
					value = ISR_oem_consolidation_score
					compare = greater_than_or_equals
				}
			}
			add_ideas = ISR_oem_startup_nation_network
		}
		else_if = {
			limit = {
				check_variable = {
					var = ISR_oem_national_champion_score
					value = ISR_oem_consolidation_score
					compare = greater_than_or_equals
				}
			}
			add_ideas = ISR_oem_national_champions
		}
		else = {
			add_ideas = ISR_oem_strategic_consolidation
		}
	}
}

# A milestone crossed after campaign start remains a choice instead of silent history.
ISR_oem_schedule_current_year_event = {
	if = {
		limit = {
			date > 2000.5.29
			has_start_date < 2000.5.30
			NOT = { has_country_flag = ISR_oem_event_1_expected }
			NOT = { has_country_flag = ISR_oem_event_1_resolved }
		}
		set_country_flag = ISR_oem_schedule_immediately
		ISR_oem_schedule_event_1 = yes
		clr_country_flag = ISR_oem_schedule_immediately
	}
	else_if = {
		limit = {
			date > 2000.11.15
			has_start_date < 2000.11.16
			NOT = { check_variable = { ISR_oem_event_1_route = 0 } }
			NOT = { has_country_flag = ISR_oem_event_2_expected }
			NOT = { has_country_flag = ISR_oem_event_2_resolved }
		}
		set_country_flag = ISR_oem_schedule_immediately
		ISR_oem_schedule_event_2 = yes
		clr_country_flag = ISR_oem_schedule_immediately
	}
	else_if = {
		limit = {
			date > 2001.3.1
			has_start_date < 2001.3.2
			NOT = { check_variable = { ISR_oem_event_2_route = 0 } }
			NOT = { has_country_flag = ISR_oem_event_3_expected }
			NOT = { has_country_flag = ISR_oem_event_3_resolved }
		}
		set_country_flag = ISR_oem_schedule_immediately
		ISR_oem_schedule_event_3 = yes
		clr_country_flag = ISR_oem_schedule_immediately
	}
	else_if = {
		limit = {
			date > 2001.7.31
			has_start_date < 2001.8.1
			NOT = { check_variable = { ISR_oem_event_3_route = 0 } }
			NOT = { has_country_flag = ISR_oem_event_4_expected }
			NOT = { has_country_flag = ISR_oem_event_4_resolved }
		}
		set_country_flag = ISR_oem_schedule_immediately
		ISR_oem_schedule_event_4 = yes
		clr_country_flag = ISR_oem_schedule_immediately
	}
}

# Recovery redelivers once after the normal pending window expires.
ISR_oem_recover_interrupted_events = {
	if = {
		limit = {
			date > 2000.5.29
			has_country_flag = ISR_oem_event_1_expected
			NOT = { has_country_flag = ISR_oem_event_1_pending }
			NOT = { has_country_flag = ISR_oem_event_1_resolved }
			NOT = { has_country_flag = ISR_oem_event_1_recovery_used }
			check_variable = { ISR_oem_event_1_route = 0 }
		}
		set_country_flag = ISR_oem_event_1_recovery_used
		country_event = ISR_oem_events.1
	}
	else_if = {
		limit = {
			date > 2000.11.15
			has_country_flag = ISR_oem_event_2_expected
			NOT = { has_country_flag = ISR_oem_event_2_pending }
			NOT = { has_country_flag = ISR_oem_event_2_resolved }
			NOT = { has_country_flag = ISR_oem_event_2_recovery_used }
			NOT = { check_variable = { ISR_oem_event_1_route = 0 } }
			check_variable = { ISR_oem_event_2_route = 0 }
		}
		set_country_flag = ISR_oem_event_2_recovery_used
		country_event = ISR_oem_events.2
	}
	else_if = {
		limit = {
			date > 2001.3.1
			has_country_flag = ISR_oem_event_3_expected
			NOT = { has_country_flag = ISR_oem_event_3_pending }
			NOT = { has_country_flag = ISR_oem_event_3_resolved }
			NOT = { has_country_flag = ISR_oem_event_3_recovery_used }
			NOT = { check_variable = { ISR_oem_event_2_route = 0 } }
			check_variable = { ISR_oem_event_3_route = 0 }
		}
		set_country_flag = ISR_oem_event_3_recovery_used
		country_event = ISR_oem_events.3
	}
	else_if = {
		limit = {
			date > 2001.7.31
			has_country_flag = ISR_oem_event_4_expected
			NOT = { has_country_flag = ISR_oem_event_4_pending }
			NOT = { has_country_flag = ISR_oem_event_4_resolved }
			NOT = { has_country_flag = ISR_oem_event_4_recovery_used }
			NOT = { check_variable = { ISR_oem_event_3_route = 0 } }
			check_variable = { ISR_oem_event_4_route = 0 }
		}
		set_country_flag = ISR_oem_event_4_recovery_used
		country_event = ISR_oem_events.4
	}
}

ISR_oem_monthly_driver = {
	if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = ISR
			NOT = { has_country_flag = collapsed_nation }
			NOT = {
				OR = {
					has_idea = ISR_oem_startup_nation_network
					has_idea = ISR_oem_national_champions
					has_idea = ISR_oem_strategic_consolidation
				}
			}
		}
		if = {
			limit = { NOT = { has_country_flag = ISR_oem_history_initialized } }
			ISR_oem_initialize_history = yes
			if = {
				limit = { corporate_history_full_enabled = yes }
				ISR_oem_reconstruct_start_history = yes
			}
		}
		if = {
			limit = { corporate_history_outcomes_only_enabled = yes }
			ISR_oem_reconstruct_reached_history = yes
		}
		else = {
			ISR_oem_schedule_current_year_event = yes
			ISR_oem_recover_interrupted_events = yes
		}
		ISR_oem_resolve_strategy = yes
	}
}
