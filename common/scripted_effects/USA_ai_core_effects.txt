USA_ai_core_resolve_capstone = {
	USA_ai_core_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_ai_core_capstone_resolved } }
		if = {
			limit = { check_variable = { USA_ai_core_capstone_choice = 1 } }
			USA_ai_core_apply_private_frontier_primacy = yes
		}
		else_if = {
			limit = { check_variable = { USA_ai_core_capstone_choice = 2 } }
			USA_ai_core_apply_open_innovation_ecosystem = yes
		}
		else_if = {
			limit = { check_variable = { USA_ai_core_capstone_choice = 3 } }
			USA_ai_core_apply_federal_compute_compact = yes
		}
		else_if = {
			limit = { check_variable = { USA_ai_core_capstone_choice = 4 } }
			USA_ai_core_apply_resilient_industrial_base = yes
		}
		else_if = {
			limit = { check_variable = { USA_ai_core_capstone_choice = 5 } }
			USA_ai_core_apply_power_constrained_leadership = yes
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = USA_ai_core_infrastructure_burden value = 8 compare = greater_than_or_equals }
					check_variable = { energy_balance < 0 }
					check_variable = { unfulfilled_energy_demand_var > 0 }
				}
			}
			USA_ai_core_apply_power_constrained_leadership = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_ai_core_ecosystem_openness value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_ai_core_public_legitimacy value = 7 compare = greater_than_or_equals }
			}
			USA_ai_core_apply_open_innovation_ecosystem = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_ai_core_state_alignment value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_ai_core_compute_depth value = 7 compare = greater_than_or_equals }
			}
			USA_ai_core_apply_federal_compute_compact = yes
		}
		else_if = {
			limit = { check_variable = { var = USA_ai_core_frontier_capability value = 8 compare = greater_than_or_equals } }
			USA_ai_core_apply_private_frontier_primacy = yes
		}
		else = { USA_ai_core_apply_resilient_industrial_base = yes }
	}
}
