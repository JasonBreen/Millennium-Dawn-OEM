# Sony corporate history. Five milestones set the route; the closing milestone
# resolves it into one of four futures for Japanese consumer electronics.

JAP_sony_apply_one_sony_ecosystem = {
	if = {
		limit = { JAP_sony_outcome_settled = no }
		add_ideas = JAP_sony_one_sony_ecosystem
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
JAP_sony_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = JAP_electronics.300 days = 272 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = JAP_sony_ericsson_venture
	}
}
