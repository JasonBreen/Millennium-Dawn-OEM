DEUSEX_monthly_china_pulse = {
	if = {
		limit = { DEUSEX_scenario_enabled = yes country_exists = CHI }
		CHI = {
			if = {
				limit = {
					check_variable = { DEUSEX_exchange_phase = 1 }
					OR = {
						NOT = { country_exists = USA }
						has_war_with = USA
					}
				}
				set_variable = { DEUSEX_exchange_phase = 4 }
				country_event = { id = DEUSEX.13 days = 1 }
			}
		}
	}
}
