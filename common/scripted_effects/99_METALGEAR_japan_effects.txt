METALGEAR_monthly_japan_pulse = {
	if = {
		limit = {
			country_exists = JAP
			JAP = {
				check_variable = { METALGEAR_request_phase = 1 }
				OR = {
					NOT = { country_exists = SOV }
					has_war_with = SOV
				}
			}
		}
		JAP = {
			set_variable = { METALGEAR_request_phase = 4 }
			country_event = { id = METALGEAR.23 days = 1 }
		}
	}
}
