# Huawei corporate history. Five milestones set the route; the closing
# milestone resolves it into one of four outcomes.

CHI_huawei_apply_sovereign_full_stack = {
	if = {
		limit = { CHI_huawei_outcome_settled = no }
		add_ideas = CHI_huawei_sovereign_full_stack
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
CHI_huawei_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = CHI_semiconductor_events.300 days = 90 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = CHI_huawei_international_expansion
	}
}
