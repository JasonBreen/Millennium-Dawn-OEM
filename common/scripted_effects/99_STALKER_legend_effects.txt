# Zone 1 dates. 2011 sets one of STALKER_strelok_marked, STALKER_strelok_turned_back,
# or STALKER_strelok_remembers. Fairway reads the 2012 route four months later.
# global.STALKER_fairway_result is 1 evacuate, 2 foothold, 3 bury, 4 massacre.
# global.STALKER_sircaa_result is 1 charter, 2 army.

STALKER_schedule_fairway = {
	if = {
		limit = { NOT = { has_global_flag = STALKER_fairway_scheduled } }
		set_global_flag = STALKER_fairway_scheduled
		set_variable = { global.STALKER_legend_deliver = 174 }
		if = {
			limit = { country_exists = USA }
			USA = { country_event = { id = STALKER.178 days = 120 } }
		}
		else = { country_event = { id = STALKER.178 days = 120 } }
	}
}

STALKER_apply_clear_sky_spared = {
	if = {
		limit = { NOT = { has_global_flag = STALKER_2011_resolved } }
		set_global_flag = STALKER_2011_resolved
		set_global_flag = STALKER_clear_sky_spared
		set_global_flag = STALKER_strelok_turned_back
		set_global_flag = STALKER_scar_survived
		log = "[GetDateText]: STALKER 2011, Clear Sky turned Strelok back"
		news_event = STALKER.173
	}
}

STALKER_apply_clear_sky_canon = {
	if = {
		limit = { NOT = { has_global_flag = STALKER_2011_resolved } }
		set_global_flag = STALKER_2011_resolved
		set_global_flag = STALKER_strelok_marked
		set_global_flag = STALKER_scar_survived
		log = "[GetDateText]: STALKER 2011, Clear Sky fell and Strelok was marked"
		698 = {
			if = {
				limit = { STALKER_is_active_zone_anchor = yes }
				set_variable = { STALKER_zone_clear_sky = 0 }
				STALKER_clamp_zone_society = yes
				if = {
					limit = { controller = { exists = yes } }
					controller = { STALKER_refresh_zone_administration = yes }
				}
			}
		}
		news_event = STALKER.173
	}
}

STALKER_apply_fairway_evacuate = {
	if = {
		limit = { NOT = { has_variable = global.STALKER_fairway_result } }
		set_variable = { global.STALKER_fairway_result = 1 }
		log = "[GetDateText]: STALKER Fairway evacuated the scientists"
		698 = {
			if = {
				limit = { STALKER_is_active_zone_anchor = yes }
				if = {
					limit = { check_variable = { global.STALKER_2012_route = 1 } }
					add_to_variable = { STALKER_zone_scientists = 8 }
					add_to_variable = { STALKER_zone_military = 5 }
					add_to_variable = { STALKER_zone_containment = 5 }
				}
				else = {
					add_to_variable = { STALKER_zone_scientists = 5 }
					add_to_variable = { STALKER_zone_military = -5 }
					add_to_variable = { STALKER_zone_containment = 5 }
				}
				clamp_variable = { var = STALKER_zone_containment min = 0 max = 100 }
				STALKER_clamp_zone_society = yes
				if = {
					limit = { controller = { exists = yes } }
					controller = { STALKER_refresh_zone_administration = yes }
				}
			}
		}
		news_event = STALKER.175
	}
}

STALKER_raise_ward_to_charter = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = STALKER_zone_ward }
				check_variable = { STALKER_zone_ward < 15 }
			}
		}
		set_variable = { STALKER_zone_ward = 15 }
	}
	if = {
		limit = { has_global_flag = STALKER_scar_survived }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = STALKER_zone_spark }
					check_variable = { STALKER_zone_spark < 10 }
				}
			}
			set_variable = { STALKER_zone_spark = 10 }
		}
	}
}

STALKER_apply_sircaa_charter = {

}
