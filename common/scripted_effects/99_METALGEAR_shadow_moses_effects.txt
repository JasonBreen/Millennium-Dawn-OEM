METALGEAR_dispatch_shadow_moses = {
	if = {
		limit = {
			METALGEAR_scenario_enabled = yes
			NOT = { has_global_flag = METALGEAR_shadow_moses_started }
		}
		set_global_flag = METALGEAR_shadow_moses_started
		if = {
			limit = {
				country_exists = USA
				814 = { is_controlled_by = USA }
			}
			USA = {
				set_variable = { METALGEAR_crisis_phase = 1 }
				country_event = { id = METALGEAR.10 days = 1 }
			}
		}
		else = {
			log = "[GetDateText]: METALGEAR Shadow Moses unavailable: United States or Alaska control missing"
			set_variable = { global.METALGEAR_shadow_moses_result = 4 }
		}
	}
}

METALGEAR_authorize_shadow_moses = {
	if = {
		limit = { check_variable = { METALGEAR_review_phase > 2 } }
		add_political_power = -25
	}
	else = { add_political_power = -50 }
}

METALGEAR_begin_shadow_moses_response = {
	set_variable = { METALGEAR_crisis_phase = 2 }
	if = {
		limit = { country_exists = JAP }
		JAP = { country_event = { id = METALGEAR.22 days = 1 } }
	}
	if = {
		limit = { country_exists = SOV }
		SOV = { country_event = { id = METALGEAR.32 days = 1 } }
	}
	country_event = { id = METALGEAR.12 days = 30 }
}

METALGEAR_resolve_shadow_moses = {
	if = {
		limit = {
			check_variable = { METALGEAR_crisis_phase = 2 }
			check_variable = { global.METALGEAR_shadow_moses_result < 1 }
		}
		if = {
			limit = { 814 = { is_controlled_by = USA } }
			set_variable = { METALGEAR_crisis_phase = 3 }
			if = {
				limit = {
					check_variable = { METALGEAR_review_phase > 2 }
					NOT = { check_variable = { METALGEAR_crisis_route = 3 } }
				}
				set_variable = { global.METALGEAR_shadow_moses_result = 1 }
				add_stability = -0.01
			}
			else = {
				set_variable = { global.METALGEAR_shadow_moses_result = 2 }
				add_stability = -0.02
			}
		}
		else = {
			set_variable = { global.METALGEAR_shadow_moses_result = 3 }
			set_variable = { METALGEAR_crisis_phase = 3 }
			METALGEAR_send_shadow_moses_reports = yes
		}
	}
}

METALGEAR_complete_shadow_moses = {
	set_variable = { METALGEAR_crisis_phase = 4 }
	METALGEAR_send_shadow_moses_reports = yes
	news_event = { id = METALGEAR.13 days = 1 }
}

METALGEAR_send_shadow_moses_reports = {
	if = {
		limit = { country_exists = JAP }
		JAP = {
			if = {
				limit = { check_variable = { METALGEAR_crisis_records < 1 } }
				set_variable = { METALGEAR_crisis_records = 3 }
			}
			country_event = { id = METALGEAR.25 days = 1 }
		}
	}
	if = {
		limit = { country_exists = SOV }
		SOV = {
			if = {
				limit = { check_variable = { METALGEAR_crisis_records < 1 } }
				set_variable = { METALGEAR_crisis_records = 3 }
			}
			country_event = { id = METALGEAR.35 days = 1 }
		}
	}
}

METALGEAR_interrupt_shadow_moses = {
	if = {
		limit = { check_variable = { global.METALGEAR_shadow_moses_result < 1 } }
		set_variable = { global.METALGEAR_shadow_moses_result = 3 }
		if = {
			limit = { country_exists = USA }
			USA = {
				set_variable = { METALGEAR_crisis_phase = 3 }
				country_event = { id = METALGEAR.8 days = 1 }
			}
		}
		METALGEAR_send_shadow_moses_reports = yes
	}
}

METALGEAR_monthly_shadow_moses_pulse = {
	if = {
		limit = {
			METALGEAR_scenario_enabled = yes
			has_global_flag = METALGEAR_shadow_moses_started
			check_variable = { global.METALGEAR_shadow_moses_result < 1 }
			OR = {
				NOT = { country_exists = USA }
				814 = { NOT = { is_controlled_by = USA } }
			}
		}
		METALGEAR_interrupt_shadow_moses = yes
	}
}

METALGEAR_share_shadow_moses_report = {
	if = {
		limit = {
			country_exists = JAP
			NOT = { has_war_with = JAP }
			JAP = { check_variable = { METALGEAR_crisis_records = 1 } }
		}
		add_opinion_modifier = { target = JAP modifier = METALGEAR_shadow_moses_disclosure }
		JAP = { add_opinion_modifier = { target = USA modifier = METALGEAR_shadow_moses_disclosure } }
	}
	if = {
		limit = {
			country_exists = SOV
			NOT = { has_war_with = SOV }
			SOV = { check_variable = { METALGEAR_crisis_records = 1 } }
		}
		add_opinion_modifier = { target = SOV modifier = METALGEAR_shadow_moses_disclosure }
		SOV = { add_opinion_modifier = { target = USA modifier = METALGEAR_shadow_moses_disclosure } }
	}
}

METALGEAR_suppress_shadow_moses_report = {
	if = {
		limit = {
			country_exists = JAP

		}
		add_opinion_modifier = { target = JAP modifier = diplomatic_insults }
		JAP = { add_opinion_modifier = { target = USA modifier = diplomatic_insults } }
	}
	if = {
		limit = {
			country_exists = SOV
			NOT = { has_war_with = SOV }
			SOV = { check_variable = { METALGEAR_crisis_records = 1 } }
		}
		add_opinion_modifier = { target = SOV modifier = diplomatic_insults }
		SOV = { add_opinion_modifier = { target = USA modifier = diplomatic_insults } }
	}
}
