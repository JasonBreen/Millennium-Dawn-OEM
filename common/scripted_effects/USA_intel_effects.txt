USA_intel_update_capstone = {
	USA_intel_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_intel_history_complete } }
		USA_intel_clear_capstone_outcomes = yes
		USA_intel_advance_era_6 = yes
		if = {
			limit = {
				has_country_flag = USA_intel_us_equity_national_anchor
				has_country_flag = USA_intel_18a_domestic_ramp
				check_variable = { var = USA_intel_process_execution value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_manufacturing_resilience value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_foundry_credibility value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_capital_discipline value = 6 compare = greater_than_or_equals }
			}
			USA_intel_apply_domestic_silicon_arsenal = yes
		}
		else_if = {
			limit = {
				has_country_flag = USA_intel_18a_open_foundry_launch
				check_variable = { var = USA_intel_foundry_credibility value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_platform_leverage value = 6 compare = less_than_or_equals }
				check_variable = { var = USA_intel_portfolio_breadth value = 8 compare = greater_than_or_equals }
			}
			USA_intel_apply_open_systems_foundry = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_intel_architecture_leadership value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_foundry_credibility value = 6 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_portfolio_breadth value = 8 compare = greater_than_or_equals }
			}
			USA_intel_apply_integrated_compute_foundry = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_intel_architecture_leadership value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_platform_leverage value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_foundry_credibility value = 6 compare = less_than }
			}
			USA_intel_apply_x86_platform_fortress = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_intel_architecture_leadership value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_intel_foundry_credibility value = 3 compare = less_than_or_equals }
				check_variable = { var = USA_intel_manufacturing_resilience value = 6 compare = less_than_or_equals }
			}
			USA_intel_apply_fabless_architecture_house = yes
		}
		else = { USA_intel_apply_managed_retrenchment = yes }

		USA_intel_clamp_state = yes
		USA_physical_compute_stack_resolve = yes
	}
}
