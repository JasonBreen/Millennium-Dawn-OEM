# Treaty observers who reach a station during a polar artifact study seize the samples and report them.
STALKER_polar_study_detected = {
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_polar_artifact_samples } }
		remove_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
	}
	set_country_flag = { flag = STALKER_polar_study_cut_short days = 180 value = 1 }
	news_event = STALKER.191
}

# A study stops when its Radiation Laboratory does: damage, lost fuel or power, a replaced module or a dismantled
# station. The samples are lost and the result never comes.
STALKER_monthly_antarctic_pulse = {
	every_country = {
		limit = {
			has_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
			NOT = { STALKER_has_working_radiation_lab = yes }
		}
		log = "[GetDateText]: [THIS.GetName]: STALKER polar artifact study ended, laboratory down"
		remove_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
		set_country_flag = { flag = STALKER_polar_study_cut_short days = 180 value = 1 }
	}
}
