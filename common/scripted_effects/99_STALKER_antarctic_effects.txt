# Treaty observers who reach a station during a polar artifact study seize the samples and report them.
STALKER_polar_study_detected = {
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_polar_artifact_samples } }
		remove_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
	}
	set_country_flag = { flag = STALKER_polar_study_seized days = 180 value = 1 }
	news_event = STALKER.191
}
