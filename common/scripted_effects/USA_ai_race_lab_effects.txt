# USA scope, from the AI Race monthly work. Each lab event fires once, after its date, while Corporate History Full
# runs that lab's chain. At most one fires a month, so they never stack.
USA_ai_race_lab_pulse = {
	if = {
		limit = {
			corporate_history_full_enabled = yes
			NOT = { has_country_flag = collapsed_nation }
		}
		set_temp_variable = { usa_lab_fired = 0 }
		USA_ai_race_lab_try_palantir = yes
		USA_ai_race_lab_try_openai = yes
		USA_ai_race_lab_try_anthropic = yes
	}
}

USA_ai_race_lab_try_palantir = {
	if = {
		limit = {
			check_variable = { usa_lab_fired = 0 }
			has_country_flag = USA_palantir_corporate_state_initialized
		}
		if = {
			limit = {
				date > 2024.10.31
				NOT = { has_country_flag = USA_ai_race_lab_1_fired }
			}
			set_country_flag = USA_ai_race_lab_1_fired
			set_temp_variable = { usa_lab_fired = 1 }
			country_event = USA_ai_race_lab.1
		}
		else_if = {
			limit = {
				date > 2025.3.31
				NOT = { has_country_flag = USA_ai_race_lab_2_fired }
			}
			set_country_flag = USA_ai_race_lab_2_fired
			set_temp_variable = { usa_lab_fired = 1 }
			country_event = USA_ai_race_lab.2
		}
		else_if = {
			limit = {
				date > 2025.4.13
				NOT = { has_country_flag = USA_ai_race_lab_3_fired }
			}
			set_country_flag = USA_ai_race_lab_3_fired
			set_temp_variable = { usa_lab_fired = 1 }
			country_event = USA_ai_race_lab.3
		}
	}
}
