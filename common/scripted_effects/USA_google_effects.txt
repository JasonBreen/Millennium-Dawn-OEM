USA_google_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_google_state_initialized } }

		set_variable = { USA_google_workforce_consent = 8 }
		set_variable = { USA_google_infrastructure_depth = 0 }
		set_variable = { USA_google_sovereign_posture = 0 }
		set_country_flag = USA_google_state_initialized
	}

	USA_google_clamp_state = yes
}

USA_google_clamp_state = {
	clamp_variable = { var = USA_google_workforce_consent min = 0 max = 10 }
	clamp_variable = { var = USA_google_infrastructure_depth min = 0 max = 10 }
	clamp_variable = { var = USA_google_sovereign_posture min = -10 max = 10 }
}
