USA_corporate_systems_clear_economic_bridge_ideas = {
	remove_ideas = {
		USA_corporate_systems_economic_integration_1
		USA_corporate_systems_economic_integration_2
		USA_corporate_systems_economic_integration_3
		USA_corporate_systems_economic_integration_4
		USA_corporate_systems_economic_integration_5
	}
}

USA_corporate_systems_update_economic_bridge = {
	if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = USA
			NOT = { has_country_flag = collapsed_nation }
			has_country_flag = USA_ibm_state_initialized
			USA_corporate_systems_has_meaningful_state = yes
		}
		set_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_open_standards }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_vertical_integration }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_supply_resilience }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_security_control }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_national_compute_stack }
		if = {
			limit = {
				has_country_flag = USA_stack_physical_capstone_resolved
				check_variable = { USA_corporate_systems_economic_integration_score < 50 }
			}
			add_to_temp_variable = { USA_corporate_systems_economic_integration_score = 1 }
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = USA_stack_physical_capstone_resolved }
				has_country_flag = USA_ti_capstone_resolved
				has_country_flag = USA_micron_capstone_resolved
				has_country_flag = USA_stack_motorola_capstone_resolved
				check_variable = { USA_corporate_systems_economic_integration_score > 0 }
			}
			subtract_from_temp_variable = { USA_corporate_systems_economic_integration_score = 1 }
		}

		# Bound platform and cloud maturity to one integration point.
		if = {
			limit = {
				has_country_flag = USA_oracle_event_12_resolved
				check_variable = { USA_corporate_systems_economic_integration_score < 50 }
				OR = {
					check_variable = { USA_oracle_platform_scale > 8 }
					check_variable = { USA_oracle_execution_discipline > 5 }
					check_variable = { USA_oracle_infrastructure_depth > 5 }
				}
			}
			add_to_temp_variable = { USA_corporate_systems_economic_integration_score = 1 }
		}

		# Bound structural-separation damage to one point and never below zero.
		if = {
			limit = {
				has_country_flag = USA_google_antitrust_structural_separation
				check_variable = { USA_google_platform_scale < 0 }
				check_variable = { USA_corporate_systems_economic_integration_score > 0 }
			}
			subtract_from_temp_variable = { USA_corporate_systems_economic_integration_score = 1 }
		}

		if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 15 } }
			if = {
				limit = {
					OR = {
						NOT = { has_idea = USA_corporate_systems_economic_integration_1 }
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_1
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 22 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						NOT = { has_idea = USA_corporate_systems_economic_integration_2 }
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_2
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 29 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						NOT = { has_idea = USA_corporate_systems_economic_integration_3 }
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_3
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 38 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						NOT = { has_idea = USA_corporate_systems_economic_integration_4 }
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_4
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						NOT = { has_idea = USA_corporate_systems_economic_integration_5 }
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_5
			}
		}
	}
	else = {
		if = {
			limit = {
				OR = {
					has_idea = USA_corporate_systems_economic_integration_1
					has_idea = USA_corporate_systems_economic_integration_2
					has_idea = USA_corporate_systems_economic_integration_3
					has_idea = USA_corporate_systems_economic_integration_4
					has_idea = USA_corporate_systems_economic_integration_5
				}
			}
			USA_corporate_systems_clear_economic_bridge_ideas = yes
		}
	}
}
