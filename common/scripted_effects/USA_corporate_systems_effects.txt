# National economic bridge. Mapping table: .claude/docs/usa-oem-economic-bridge-map.md
#
# Layering: the five USA_oem_* axes are the base state written by the IBM and
# Sun/Microsoft chains. Every other mature chain contributes through the
# USA_oem_contribution_* accumulators below, which are rebuilt from scratch on
# every bridge call, capped per axis, and added to the base to produce the
# USA_oem_effective_* axes the national score and the dashboard read.

USA_corporate_systems_clear_economic_bridge_ideas = {
	remove_ideas = {
		USA_corporate_systems_economic_integration_1
		USA_corporate_systems_economic_integration_2
		USA_corporate_systems_economic_integration_3
		USA_corporate_systems_economic_integration_4
		USA_corporate_systems_economic_integration_5
	}
}

	set_variable = { USA_oem_contribution_vertical_integration = 0 }
	set_variable = { USA_oem_contribution_supply_resilience = 0 }

	set_variable = { USA_oem_effective_vertical_integration = 0 }
	set_variable = { USA_oem_effective_supply_resilience = 0 }
	set_variable = { USA_oem_effective_security_control = 0 }

}

USA_corporate_systems_apple_contribution = {
	if = {
		limit = { has_country_flag = USA_apple_outcome_private_computing }
		add_to_variable = { USA_oem_contribution_vertical_integration = 1 }
		add_to_variable = { USA_oem_contribution_supply_resilience = 1 }
	}
	else_if = {
		limit = { has_country_flag = USA_apple_outcome_integrated_device }
		add_to_variable = { USA_oem_contribution_vertical_integration = 1 }
		subtract_from_variable = { USA_oem_contribution_open_standards = 1 }
	}
	else_if = {
		limit = { has_country_flag = USA_apple_outcome_resilient_silicon }
		add_to_variable = { USA_oem_contribution_supply_resilience = 1 }
	}
	else_if = {
		limit = { has_country_flag = USA_apple_outcome_regulated_services }
		add_to_variable = { USA_oem_contribution_open_standards = 1 }
		subtract_from_variable = { USA_oem_contribution_vertical_integration = 1 }
	}
}

# The three depth variables only accumulate in a played-through Full campaign;
# USA_oracle_cloud_public_sector is the reconstruction's equivalent maturity
# marker, so both modes reach the same contribution for the same history.
USA_corporate_systems_oracle_contribution = {
	if = {
		limit = {
			has_country_flag = USA_oracle_event_12_resolved
			OR = {
				has_country_flag = USA_oracle_cloud_public_sector
				check_variable = { USA_oracle_platform_scale > 8 }
				check_variable = { USA_oracle_execution_discipline > 5 }
				check_variable = { USA_oracle_infrastructure_depth > 5 }
			}
		}
		add_to_variable = { USA_oem_contribution_vertical_integration = 1 }
	}
	if = {
		limit = { has_country_flag = USA_oracle_java_api_victory }
		subtract_from_variable = { USA_oem_contribution_open_standards = 1 }
	}
	else_if = {
		limit = { has_country_flag = USA_oracle_sun_commons }
		add_to_variable = { USA_oem_contribution_open_standards = 1 }
	}
}

# Rebuilt from zero on every call, so repeated monthly ticks, reloads and
# Outcomes Only catch-up all converge on the same totals.
USA_corporate_systems_rebuild_company_contributions = {
	set_variable = { USA_oem_contribution_open_standards = 0 }
	set_variable = { USA_oem_contribution_vertical_integration = 0 }
	set_variable = { USA_oem_contribution_supply_resilience = 0 }
	set_variable = { USA_oem_contribution_security_control = 0 }
	set_variable = { USA_oem_contribution_national_compute_stack = 0 }

	USA_corporate_systems_apple_contribution = yes
	USA_corporate_systems_hp_contribution = yes
	USA_corporate_systems_nvidia_contribution = yes
	USA_corporate_systems_dell_contribution = yes
	USA_corporate_systems_ti_contribution = yes
	USA_corporate_systems_micron_contribution = yes
	USA_corporate_systems_motorola_contribution = yes
	USA_corporate_systems_google_contribution = yes
	USA_corporate_systems_oracle_contribution = yes

	clamp_variable = { var = USA_oem_contribution_open_standards min = -3 max = 3 }
	clamp_variable = { var = USA_oem_contribution_vertical_integration min = -3 max = 3 }
	clamp_variable = { var = USA_oem_contribution_supply_resilience min = -3 max = 3 }
	clamp_variable = { var = USA_oem_contribution_security_control min = -3 max = 3 }
	clamp_variable = { var = USA_oem_contribution_national_compute_stack min = -3 max = 3 }
}

USA_corporate_systems_update_economic_bridge = {
	if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = USA
			NOT = { has_country_flag = collapsed_nation }
			has_country_flag = USA_ibm_state_initialized
			USA_corporate_systems_has_meaningful_state = yes
		}
		USA_corporate_systems_rebuild_company_contributions = yes
		USA_corporate_systems_rebuild_effective_axes = yes

		set_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_open_standards }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_vertical_integration }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_supply_resilience }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_security_control }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_national_compute_stack }

		if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 15 } }
			if = {
				limit = {
					OR = {
						NOT = { has_idea = USA_corporate_systems_economic_integration_1 }
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_1
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 22 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						NOT = { has_idea = USA_corporate_systems_economic_integration_2 }
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_2
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 29 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						NOT = { has_idea = USA_corporate_systems_economic_integration_3 }
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_3
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 38 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						NOT = { has_idea = USA_corporate_systems_economic_integration_4 }
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_4
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						NOT = { has_idea = USA_corporate_systems_economic_integration_5 }
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_5
			}
		}
	}
	else = {
		# Nothing to clear until the bridge has run at least once, which cannot
		# happen before IBM initializes.
		if = {
			limit = { has_country_flag = USA_ibm_state_initialized }
			USA_corporate_systems_clear_derived_axes = yes
		}
		if = {
			limit = {
				OR = {
					has_idea = USA_corporate_systems_economic_integration_1
					has_idea = USA_corporate_systems_economic_integration_2
					has_idea = USA_corporate_systems_economic_integration_3
					has_idea = USA_corporate_systems_economic_integration_4
					has_idea = USA_corporate_systems_economic_integration_5
				}
			}
			USA_corporate_systems_clear_economic_bridge_ideas = yes
		}
	}
}
