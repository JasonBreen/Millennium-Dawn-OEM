# BlackBerry corporate history events
country_event = {
	id = CAN_blackberry_events.1
	title = CAN_blackberry_events.1.t
	desc = CAN_blackberry_events.1.d
	picture = GFX_CAN_generic
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		original_tag = CAN
		NOT = { has_country_flag = collapsed_nation }
	}

	option = {
		name = CAN_blackberry_events.1.a
		log = "[GetDateText]: [This.GetName]: CAN_blackberry_events.1.a executed"
		add_political_power = 10
		set_country_flag = CAN_blackberry_private_expansion

		ai_chance = { base = 90 }
	}

	option = {
		name = CAN_blackberry_events.1.b
		log = "[GetDateText]: [This.GetName]: CAN_blackberry_events.1.b executed"
		set_temp_variable = { treasury_change = -2.00 }
		modify_treasury_effect = yes
		add_tech_bonus = {
			name = CAN_blackberry_waterloo_computing
			bonus = 0.10
			uses = 1
			category = CAT_computing_tech
		}
		set_country_flag = CAN_blackberry_waterloo_support

		ai_chance = { base = 10 }
	}
}
