country_event = {
	id = DEUSEX.3
	title = DEUSEX.3.t
	desc = DEUSEX.3.d
	picture = GFX_hospital
	is_triggered_only = yes

	trigger = {
		DEUSEX_scenario_enabled = yes
		tag = USA
		country_exists = CHI
		NOT = { has_war_with = CHI }
		CHI = { check_variable = { DEUSEX_exchange_phase = 1 } }
	}

	option = {
		name = DEUSEX.3.a
		log = "[GetDateText]: [This.GetName]: DEUSEX.3.a executed"
		custom_effect_tooltip = DEUSEX_accept_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = CHI
					NOT = { has_war_with = CHI }
					CHI = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				add_political_power = -10
				add_opinion_modifier = { target = CHI modifier = diplomatic_support }
				reverse_add_opinion_modifier = { target = CHI modifier = diplomatic_support }
				CHI = {
					set_variable = { DEUSEX_exchange_phase = 2 }
					country_event = { id = DEUSEX.13 days = 1 }
				}
			}
			else_if = {
				limit = {
					country_exists = CHI
					CHI = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				CHI = {
					set_variable = { DEUSEX_exchange_phase = 4 }
					country_event = { id = DEUSEX.13 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { add = 20 has_opinion = { target = CHI value > 30 } }
			modifier = { factor = 0.25 has_opinion = { target = CHI value < -30 } }
		}
	}

	option = {
		name = DEUSEX.3.b
		log = "[GetDateText]: [This.GetName]: DEUSEX.3.b executed"
		custom_effect_tooltip = DEUSEX_decline_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = CHI
					CHI = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				CHI = {
					if = {
						limit = { has_war_with = USA }
						set_variable = { DEUSEX_exchange_phase = 4 }
					}
					else = { set_variable = { DEUSEX_exchange_phase = 3 } }
					country_event = { id = DEUSEX.13 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_opinion = { target = CHI value < -30 } }
		}
	}
}
