# An invitation to another forum's summit. Claims the oldest queued forum and saves its host as econ_forum_inviter.
country_event = {
	id = econ_forum.2
	title = econ_forum.2.t
	desc = econ_forum.2.d
	picture = GFX_economic_forum_aze
	is_triggered_only = yes

	trigger = { check_variable = { econ_forum_invite_queue^num > 0 } }

	immediate = {
		hidden_effect = {
			set_temp_variable = { ef_i = econ_forum_invite_queue^0 }
			remove_from_array = { array = econ_forum_invite_queue index = 0 }
			var:global.econ_forum_host^ef_i = { save_event_target_as = econ_forum_inviter }
			if = {
				limit = { NOT = { check_variable = { global.econ_forum_seat^ef_i = 0 } } }
				var:global.econ_forum_seat^ef_i = { save_event_target_as = econ_forum_invite_seat }
			}
		}
	}

	option = {
		name = econ_forum.2.a
		trigger = { has_political_power > 19 }
		log = "[GetDateText]: [This.GetName]: econ_forum.2.a executed"
		add_political_power = -20
		custom_effect_tooltip = econ_forum_send_head_tt
		hidden_effect = {
			set_temp_variable = { ef_answer = 4 }
			set_temp_variable = { ef_cost = 20 }
			econ_forum_answer_invitation = yes
		}
		ai_chance = { base = 1 }
	}

	option = {
		name = econ_forum.2.b
		trigger = { has_political_power > 4 }
		log = "[GetDateText]: [This.GetName]: econ_forum.2.b executed"
		add_political_power = -5
		custom_effect_tooltip = econ_forum_send_ministers_tt
		hidden_effect = {
			set_temp_variable = { ef_answer = 3 }
			set_temp_variable = { ef_cost = 5 }
			econ_forum_answer_invitation = yes
		}
		ai_chance = { base = 2 }
	}

	option = {
		name = econ_forum.2.c
		log = "[GetDateText]: [This.GetName]: econ_forum.2.c executed"
		hidden_effect = {
			set_temp_variable = { ef_answer = -1 }
			set_temp_variable = { ef_cost = 0 }
			econ_forum_answer_invitation = yes
		}
		ai_chance = { base = 1 }
	}
}
