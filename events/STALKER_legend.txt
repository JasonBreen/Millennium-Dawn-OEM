# Waits out the delay, then gives the dated choice to whoever holds the Zone.
# global.STALKER_legend_deliver is 172 for the 2011 assault, 174 for Fairway, and 176 for SIRCAA.
country_event = {
	id = STALKER.178
	hidden = yes
	is_triggered_only = yes

	trigger = { STALKER_scenario_enabled = yes }

	immediate = {
		log = "[GetDateText]: [This.GetName]: STALKER.178 immediate executed"
		698 = {
			if = {
				limit = {
					STALKER_is_active_zone_anchor = yes
					check_variable = { STALKER_zone_id = 1 }
					controller = { exists = yes }
				}
				controller = {
					if = {
						limit = { check_variable = { global.STALKER_legend_deliver = 172 } }
						country_event = STALKER.172
					}
					else_if = {
						limit = { check_variable = { global.STALKER_legend_deliver = 176 } }
						country_event = STALKER.176
					}
					else = { country_event = STALKER.174 }
				}
			}
		}
		set_variable = { global.STALKER_legend_deliver = 0 }
	}
}

# Route 1 only. Fires the existing Noon news once Strider's people wake.
country_event = {
	id = STALKER.179
	hidden = yes
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		NOT = { has_global_flag = STALKER_strider_told }
	}

	immediate = {
		log = "[GetDateText]: [This.GetName]: STALKER.179 immediate executed"
		set_global_flag = STALKER_strider_told
		698 = {
			if = {
				limit = {
					STALKER_is_active_zone_anchor = yes
					NOT = { has_variable = STALKER_zone_noon }
					controller = { exists = yes }
				}
				add_to_variable = { STALKER_zone_noon = 10 }
				clamp_variable = { var = STALKER_zone_noon min = 0 max = 100 }
				STALKER_clamp_zone_society = yes
				STALKER_target_zone = yes
				controller = { country_event = STALKER.123 }
			}
		}
	}
}
