country_event = {
	id = DEUSEX.22
	title = DEUSEX.22.t
	desc = DEUSEX.22.d
	picture = GFX_hospital
	is_triggered_only = yes

	trigger = {
		DEUSEX_scenario_enabled = yes
		tag = GER
		country_exists = USA
		NOT = { has_war_with = USA }
		USA = { check_variable = { DEUSEX_exchange_phase = 1 } }
	}

	option = {
		name = DEUSEX.22.a
		log = "[GetDateText]: [This.GetName]: DEUSEX.22.a executed"
		custom_effect_tooltip = DEUSEX_accept_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = USA
					NOT = { has_war_with = USA }
					USA = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				add_political_power = -10
				add_opinion_modifier = { target = USA modifier = diplomatic_support }
				reverse_add_opinion_modifier = { target = USA modifier = diplomatic_support }
				USA = {
					set_variable = { DEUSEX_exchange_phase = 2 }
					country_event = { id = DEUSEX.4 days = 1 }
				}
			}
			else_if = {
				limit = {
					country_exists = USA
					USA = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				USA = {
					set_variable = { DEUSEX_exchange_phase = 4 }
					country_event = { id = DEUSEX.4 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { add = 20 has_opinion = { target = USA value > 30 } }
			modifier = { factor = 0.25 has_opinion = { target = USA value < -30 } }
		}
	}

	option = {
		name = DEUSEX.22.b
		log = "[GetDateText]: [This.GetName]: DEUSEX.22.b executed"
		custom_effect_tooltip = DEUSEX_decline_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = USA
					USA = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				USA = {
					if = {
						limit = { has_war_with = GER }
						set_variable = { DEUSEX_exchange_phase = 4 }
					}
					else = { set_variable = { DEUSEX_exchange_phase = 3 } }
					country_event = { id = DEUSEX.4 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_opinion = { target = USA value < -30 } }
		}
	}
}
