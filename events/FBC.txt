# The Unscheduled Floor. Site is New York, state 769, for the whole chain.
country_event = {
	id = FBC.1
	title = FBC.1.t
	desc = FBC.1.d
	picture = GFX_UN_emergency_meeting
	is_triggered_only = yes

	trigger = {
		FBC_scenario_enabled = yes
		original_tag = USA
		check_variable = { FBC_floor_phase = 0 }
	}

	option = {
		name = FBC.1.a
		log = "[GetDateText]: [This.GetName]: FBC.1.a executed"
		custom_effect_tooltip = FBC.1.a_tt
		trigger = {
			check_variable = { FBC_response_capacity > 0 }
			769 = { is_controlled_by = ROOT }
		}
		hidden_effect = {
			if = {
				limit = {
					check_variable = { FBC_floor_phase = 0 }
					check_variable = { FBC_response_capacity > 0 }
					769 = { is_controlled_by = ROOT }
				}
				FBC_commit_floor_team = yes
				set_variable = { FBC_floor_phase = 2 }
				country_event = { id = FBC.2 days = 40 }
			}
		}
		ai_chance = { base = 50 }
	}

	option = {
		name = FBC.1.b
		log = "[GetDateText]: [This.GetName]: FBC.1.b executed"
		custom_effect_tooltip = FBC.1.b_tt
		trigger = {
			check_variable = { FBC_response_capacity > 0 }
			769 = { is_controlled_by = ROOT }
		}
		hidden_effect = {
			if = {
				limit = {
					check_variable = { FBC_floor_phase = 0 }
					check_variable = { FBC_response_capacity > 0 }
					769 = { is_controlled_by = ROOT }
				}
				FBC_commit_floor_team = yes
				set_variable = { FBC_floor_phase = 3 }
				add_to_variable = { FBC_exposure = -3 }
				FBC_clamp_bureau = yes
				FBC_refresh_bureau_factors = yes
				add_stability = -0.01
				769 = { set_state_flag = { flag = FBC_floor_lockdown value = 1 days = 120 } }
				country_event = { id = FBC.3 days = 50 }
			}
		}
		ai_chance = { base = 30 }
	}

	option = {
		name = FBC.1.c
		log = "[GetDateText]: [This.GetName]: FBC.1.c executed"
		custom_effect_tooltip = FBC.1.c_tt
		trigger = { 769 = { is_controlled_by = ROOT } }
		hidden_effect = {
			if = {
				limit = {
					check_variable = { FBC_floor_phase = 0 }
					769 = { is_controlled_by = ROOT }
				}
				set_variable = { FBC_floor_phase = 1 }
				add_to_variable = { FBC_exposure = 8 }
				FBC_clamp_bureau = yes
				FBC_refresh_bureau_factors = yes
				country_event = { id = FBC.2 days = 70 }
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				factor = 0
				check_variable = { FBC_exposure > 30 }
			}
		}
	}

	option = {
		name = FBC.1.e
		log = "[GetDateText]: [This.GetName]: FBC.1.e executed"
		custom_effect_tooltip = FBC.1.e_tt
		trigger = { 769 = { NOT = { is_controlled_by = ROOT } } }

country_event = {
	id = FBC.2
	title = FBC.2.t
	desc = FBC.2.d
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	trigger = {
		FBC_scenario_enabled = yes
		original_tag = USA
		OR = {
			check_variable = { FBC_floor_phase = 1 }
			check_variable = { FBC_floor_phase = 2 }
		}
	}

	immediate = {
		if = {
			limit = { check_variable = { FBC_floor_phase = 1 } }
			set_variable = { FBC_floor_knowledge = 5 }
			add_to_variable = { FBC_exposure = 5 }
		}
		else_if = {
			limit = { check_variable = { FBC_floor_phase = 2 } }
			set_variable = { FBC_floor_knowledge = 15 }
		}
		FBC_clamp_bureau = yes
		FBC_refresh_bureau_factors = yes
	}

	option = {
		name = FBC.2.a
		log = "[GetDateText]: [This.GetName]: FBC.2.a executed"
		custom_effect_tooltip = FBC.2.a_tt
		trigger = {
			769 = { is_controlled_by = ROOT }
			OR = {
				check_variable = { FBC_floor_committed > 0 }
				check_variable = { FBC_response_capacity > 0 }
			}
		}
		hidden_effect = {
			if = {
				limit = {
					769 = { is_controlled_by = ROOT }
					OR = {
						check_variable = { FBC_floor_phase = 1 }
						check_variable = { FBC_floor_phase = 2 }
					}
				}
				if = {
					limit = { check_variable = { FBC_floor_committed < 1 } }
					FBC_commit_floor_team = yes
				}
				if = {
					limit = { check_variable = { FBC_floor_committed > 0 } }

		}

					check_variable = { FBC_floor_phase = 2 }
					769 = { is_controlled_by = ROOT }
				}

				random = {
					chance = 20
					set_temp_variable = { FBC_research_backfire = 1 }
				}
				if = {
					limit = { check_variable = { FBC_research_backfire = 1 } }
					add_to_variable = { FBC_exposure = 10 }
					add_to_variable = { FBC_containment_integrity = -10 }
				}
				else = {
					add_to_variable = { FBC_floor_knowledge = 10 }
					add_to_variable = { FBC_exposure = 3 }
				}
				FBC_clamp_bureau = yes
				FBC_refresh_bureau_factors = yes
				country_event = { id = FBC.3 days = 30 }
			}
		}
		ai_chance = { base = 35 }
	}

	option = {
		name = FBC.2.c
		log = "[GetDateText]: [This.GetName]: FBC.2.c executed"
		custom_effect_tooltip = FBC.2.c_tt
		trigger = { check_variable = { FBC_floor_knowledge > 14 } }
