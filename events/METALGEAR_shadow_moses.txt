country_event = {
	id = METALGEAR.12
	title = METALGEAR.12.t
	desc = {
		trigger = { check_variable = { global.METALGEAR_shadow_moses_result = 1 } }
		text = METALGEAR.12.d_prepared
	}
	desc = {
		trigger = { check_variable = { global.METALGEAR_shadow_moses_result = 2 } }
		text = METALGEAR.12.d_unsupported
	}
	desc = {
		trigger = { check_variable = { global.METALGEAR_shadow_moses_result = 3 } }
		text = METALGEAR.12.d_interrupted
	}
	picture = GFX_generic_nuclear_smuggling
	is_triggered_only = yes

	trigger = {
		METALGEAR_scenario_enabled = yes
		tag = USA
		check_variable = { METALGEAR_crisis_phase = 2 }
	}

	immediate = {
		METALGEAR_resolve_shadow_moses = yes
	}

	option = {
		name = METALGEAR.12.a
		log = "[GetDateText]: [This.GetName]: METALGEAR.12.a executed"
		custom_effect_tooltip = METALGEAR.12.a_tt
		trigger = {
			OR = {
				check_variable = { global.METALGEAR_shadow_moses_result = 1 }
				check_variable = { global.METALGEAR_shadow_moses_result = 2 }
			}
		}
		hidden_effect = {
			if = {
				limit = { check_variable = { METALGEAR_crisis_phase = 3 } }
				if = {
					limit = { check_variable = { METALGEAR_crisis_route = 2 } }
					set_variable = { METALGEAR_crisis_disclosure = 1 }
					METALGEAR_share_shadow_moses_report = yes
				}
				METALGEAR_complete_shadow_moses = yes
			}
		}

			base = 80
		}
	}

		name = METALGEAR.12.b
		log = "[GetDateText]: [This.GetName]: METALGEAR.12.b executed"
		custom_effect_tooltip = METALGEAR.12.b_tt
		trigger = {
			OR = {
				check_variable = { global.METALGEAR_shadow_moses_result = 1 }
				check_variable = { global.METALGEAR_shadow_moses_result = 2 }
			}
			check_variable = { METALGEAR_crisis_route = 2 }
		}
		hidden_effect = {
			if = {
				limit = { check_variable = { METALGEAR_crisis_phase = 3 } }
				set_variable = { METALGEAR_crisis_disclosure = 2 }
				METALGEAR_suppress_shadow_moses_report = yes
				METALGEAR_complete_shadow_moses = yes
			}
		}
		ai_chance = {
			base = 20
			modifier = { factor = 2 has_stability < 0.3 }
		}
	}

	option = {
		name = METALGEAR.12.c
		log = "[GetDateText]: [This.GetName]: METALGEAR.12.c executed"
		hidden_effect = {
			if = {
				limit = { check_variable = { METALGEAR_crisis_phase = 3 } }

			}
		}
		trigger = {
			check_variable = { global.METALGEAR_shadow_moses_result = 3 }
		}
		ai_chance = {
			base = 100
		}
	}
}
