add_namespace = METALGEAR

# The opening dossier. The player oversees the institution's response, not the operative on the ground.
country_event = {
	id = METALGEAR.10
	title = METALGEAR.10.t
	desc = METALGEAR.10.d
	picture = GFX_alaska_attack_event
	is_triggered_only = yes

	trigger = {
		METALGEAR_is_controller = yes
		NOT = { has_global_flag = METALGEAR_shadow_moses_resolved }
		check_variable = { global.METALGEAR_sm_stage = 1 }
	}

	option = {
		name = METALGEAR.10.a
		log = "[GetDateText]: [This.GetName]: METALGEAR.10.a executed"
		custom_effect_tooltip = METALGEAR.10.a_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { global.METALGEAR_sm_stage = 1 } }
				set_variable = { global.METALGEAR_sm_priority = 1 }
				METALGEAR_sm_commit = yes
				METALGEAR_sm_advance = yes
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				factor = 0.5
				check_variable = { global.METALGEAR_exposure > 40 }
			}
		}
	}

	option = {
		name = METALGEAR.10.b
		log = "[GetDateText]: [This.GetName]: METALGEAR.10.b executed"
		custom_effect_tooltip = METALGEAR.10.b_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { global.METALGEAR_sm_stage = 1 } }
				set_variable = { global.METALGEAR_sm_priority = 2 }
				METALGEAR_sm_commit = yes
				METALGEAR_sm_advance = yes
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				factor = 2
				check_variable = { global.METALGEAR_exposure > 40 }
			}
		}
	}

	option = {
		name = METALGEAR.10.c
		log = "[GetDateText]: [This.GetName]: METALGEAR.10.c executed"
		custom_effect_tooltip = METALGEAR.10.c_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { global.METALGEAR_sm_stage = 1 } }
				set_temp_variable = { treasury_change = -2 }
				modify_treasury_effect = yes
				set_variable = { global.METALGEAR_sm_priority = 3 }
				METALGEAR_sm_commit = yes
				METALGEAR_sm_advance = yes
			}
		}
		ai_chance = {
			base = 20
			modifier = {
				factor = 0
				check_variable = { treasury < 5 }
			}
		}
	}
}
