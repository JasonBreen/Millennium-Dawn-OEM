add_namespace = FBC

# The station's controller. The pulse set FBC_polar_station_id on it just before.
country_event = {
	id = FBC.20
	title = FBC.20.t
	desc = FBC.20.d
	picture = GFX_antarctic_treaty_generic_picture
	is_triggered_only = yes

	option = {
		name = FBC.20.a
		log = "[GetDateText]: [This.GetName]: FBC.20.a executed"
		custom_effect_tooltip = FBC.20.a_tt
		add_political_power = -25
		hidden_effect = {
			clear_variable = FBC_polar_station_id
			FBC_polar_close_case = yes
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = FBC.20.b
		log = "[GetDateText]: [This.GetName]: FBC.20.b executed"
		custom_effect_tooltip = FBC.20.b_tt
		hidden_effect = {
			if = {
				limit = { country_exists = USA }
				USA = { country_event = { id = FBC.21 days = 7 } }
			}
			else = {
				FBC_polar_breakout = yes
				FBC_polar_close_case = yes
			}
		}
		ai_chance = { base = 40 }
	}
}

# The Bureau. FROM is the station's controller.
country_event = {
	id = FBC.21
	title = FBC.21.t
	desc = FBC.21.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	immediate = {
		hidden_effect = { set_variable = { FBC_polar_country = FROM.id } }
	}

	option = {
		name = FBC.21.a
		log = "[GetDateText]: [This.GetName]: FBC.21.a executed"
		trigger = { check_variable = { FBC_response_capacity > 0 } }
		custom_effect_tooltip = FBC.21.a_tt
		hidden_effect = {
			FBC_commit_polar_team = yes
			random_list = {
				75 = { country_event = { id = FBC.22 days = 45 } }
				25 = { country_event = { id = FBC.24 days = 45 } }
			}
		}
		ai_chance = { base = 80 }
	}

	option = {
		name = FBC.21.b
		log = "[GetDateText]: [This.GetName]: FBC.21.b executed"
		custom_effect_tooltip = FBC.21.b_tt
		hidden_effect = {
			add_to_variable = { FBC_exposure = 5 }
			FBC_clamp_bureau = yes
			FROM = { country_event = FBC.23 }
			clear_variable = FBC_polar_country
			FBC_polar_close_case = yes
		}
		ai_chance = { base = 20 }
	}
}

# The Bureau's team contains the event.
country_event = {
	id = FBC.22
	title = FBC.22.t
	desc = FBC.22.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	option = {
		name = FBC.22.a
		log = "[GetDateText]: [This.GetName]: FBC.22.a executed"
		custom_effect_tooltip = FBC.22.a_tt
		hidden_effect = {
			add_to_variable = { FBC_containment_integrity = 5 }
			FBC_release_polar_team = yes
			FBC_clamp_bureau = yes
			if = {
				limit = { check_variable = { FBC_polar_country > 0 } }
				var:FBC_polar_country = {
					add_tech_bonus = {
						name = FBC_polar_core_findings
						bonus = 0.5
						uses = 1
						category = CAT_excavation
					}
					clear_variable = FBC_polar_station_id
				}
			}
			clear_variable = FBC_polar_country
			FBC_polar_close_case = yes
		}
		ai_chance = { base = 100 }
	}
}
