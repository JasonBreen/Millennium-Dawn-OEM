country_event = {
	id = GER_siemens_events.1
	title = GER_siemens_events.1.t
	desc = GER_siemens_events.1.d
	picture = GFX_generic_factory
	is_triggered_only = yes

	trigger = {
		corporate_history_full_enabled = yes
		original_tag = GER
		NOT = { has_country_flag = collapsed_nation }
		NOT = { has_country_flag = GER_siemens_event_1_resolved }
		NOT = { has_country_flag = GER_siemens_integrated_group }
		NOT = { has_country_flag = GER_siemens_digital_industry }
	}

	option = {
		name = GER_siemens_events.1.a
		log = "[GetDateText]: [Root.GetName]: GER_siemens_events.1.a executed"
		if = {
			limit = {
				hidden_trigger = {
					NOT = { has_country_flag = GER_siemens_integrated_group }
					NOT = { has_country_flag = GER_siemens_digital_industry }
				}
			}
			add_political_power = 5
			add_tech_bonus = {
				name = GER_siemens_events.1.integrated_bonus
				bonus = 0.10
				uses = 1
				category = CAT_cnc
			}
			GER_siemens_apply_integrated_group_outcome = yes
		}
		GER_siemens_complete_event_1 = yes
		GER_siemens_schedule_due_successor = yes
		ai_chance = {
			base = 50
			modifier = {
				factor = 10
				is_historical_focus_on = yes
			}
		}
	}

	option = {
		name = GER_siemens_events.1.b
		log = "[GetDateText]: [Root.GetName]: GER_siemens_events.1.b executed"
		if = {
			limit = {
				hidden_trigger = {
					NOT = { has_country_flag = GER_siemens_integrated_group }
					NOT = { has_country_flag = GER_siemens_digital_industry }
				}
			}
			set_temp_variable = { treasury_change = -2.00 }
			modify_treasury_effect = yes
			add_tech_bonus = {
				name = GER_siemens_events.1.digital_bonus
				bonus = 0.15
				uses = 1
				category = CAT_computing_tech
			}
			GER_siemens_apply_digital_industry_outcome = yes
		}
		GER_siemens_complete_event_1 = yes
		GER_siemens_schedule_due_successor = yes
		ai_chance = {
			base = 50
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}
