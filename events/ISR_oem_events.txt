		check_variable = { ISR_oem_event_1_route = 0 }

		name = ISR_oem_events.1.a

			limit = { hidden_trigger = { check_variable = { ISR_oem_event_1_route = 0 } } }

				name = ISR_oem_events.1.export_bonus
