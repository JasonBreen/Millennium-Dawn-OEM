# Contradictory reports from Maine. Site is state 764 for the whole case.
country_event = {
	id = FBC.30
	title = FBC.30.t
	desc = FBC.30.d
	picture = GFX_CAN_ontario_goes_dark
	is_triggered_only = yes

	trigger = {
		FBC_scenario_enabled = yes
		original_tag = USA
		check_variable = { FBC_sh_phase = 0 }
		764 = { is_controlled_by = ROOT }
	}

	immediate = {
		hidden_effect = { set_variable = { FBC_sh_phase = 1 } }
	}

	option = {
		name = FBC.30.a
		log = "[GetDateText]: [This.GetName]: FBC.30.a executed"
		trigger = {
			check_variable = { FBC_response_capacity > 0 }
			764 = { is_controlled_by = ROOT }
		}
		custom_effect_tooltip = FBC.30.a_tt
		hidden_effect = {
			if = {
				limit = {
					check_variable = { FBC_sh_phase = 1 }
					check_variable = { FBC_response_capacity > 0 }
				}
				FBC_commit_silenthill_team = yes
				set_temp_variable = { FBC_sh_choice = 1 }
				set_temp_variable = { FBC_sh_delay = 6 }
				FBC_silenthill_intervene = yes
				set_temp_variable = { FBC_sh_fog_change = -10 }
				FBC_change_silenthill_fog = yes
			}
		}
		ai_chance = { base = 50 }
	}

	option = {
		name = FBC.30.b
		log = "[GetDateText]: [This.GetName]: FBC.30.b executed"
		trigger = { 764 = { is_controlled_by = ROOT } }
		add_political_power = -50
		add_stability = -0.02
		custom_effect_tooltip = FBC.30.b_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { FBC_sh_phase = 1 } }
				set_temp_variable = { FBC_sh_choice = 2 }
				set_temp_variable = { FBC_sh_delay = 8 }
				FBC_silenthill_intervene = yes
			}
		}
		ai_chance = {
			base = 30
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { factor = 0.25 has_stability < 0.3 }
		}
	}

	option = {
		name = FBC.30.c
		log = "[GetDateText]: [This.GetName]: FBC.30.c executed"
		custom_effect_tooltip = FBC.30.c_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { FBC_sh_phase = 1 } }
				set_temp_variable = { FBC_sh_choice = 3 }
				set_temp_variable = { FBC_sh_delay = 10 }
				FBC_silenthill_intervene = yes
				add_to_variable = { FBC_exposure = -5 }
				FBC_clamp_bureau = yes
				FBC_refresh_bureau_factors = yes
			}
		}
		ai_chance = { base = 20 }
	}
