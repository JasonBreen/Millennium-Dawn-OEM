country_event = {
	id = DEUSEX.12
	title = DEUSEX.12.t
	desc = DEUSEX.12.d
	picture = GFX_hospital
	is_triggered_only = yes

	trigger = {
		DEUSEX_scenario_enabled = yes
		tag = CHI
		country_exists = GER
		NOT = { has_war_with = GER }
		GER = { check_variable = { DEUSEX_exchange_phase = 1 } }
	}

	option = {
		name = DEUSEX.12.a
		log = "[GetDateText]: [This.GetName]: DEUSEX.12.a executed"
		custom_effect_tooltip = DEUSEX_accept_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = GER
					NOT = { has_war_with = GER }
					GER = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				add_political_power = -10
				add_opinion_modifier = { target = GER modifier = diplomatic_support }
				reverse_add_opinion_modifier = { target = GER modifier = diplomatic_support }
				GER = {
					set_variable = { DEUSEX_exchange_phase = 2 }
					country_event = { id = DEUSEX.23 days = 1 }
				}
			}
			else_if = {
				limit = {
					country_exists = GER
					GER = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				GER = {
					set_variable = { DEUSEX_exchange_phase = 4 }
					country_event = { id = DEUSEX.23 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { add = 20 has_opinion = { target = GER value > 30 } }
			modifier = { factor = 0.25 has_opinion = { target = GER value < -30 } }
		}
	}

	option = {
		name = DEUSEX.12.b
		log = "[GetDateText]: [This.GetName]: DEUSEX.12.b executed"
		custom_effect_tooltip = DEUSEX_decline_exchange_tt
		hidden_effect = {
			if = {
				limit = {
					country_exists = GER
					GER = { check_variable = { DEUSEX_exchange_phase = 1 } }
				}
				GER = {
					if = {
						limit = { has_war_with = CHI }
						set_variable = { DEUSEX_exchange_phase = 4 }
					}
					else = { set_variable = { DEUSEX_exchange_phase = 3 } }
					country_event = { id = DEUSEX.23 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_opinion = { target = GER value < -30 } }
		}
	}
}
