# Full membership invitation. ROOT is the invitee, FROM the chair

	id = IMP.10
	title = IMP.10.t
	desc = IMP.10.d
	picture = GFX_belarus_intermarium
	is_triggered_only = yes

	option = {
		name = IMP.10.a
		log = "[GetDateText]: [This.GetName]: IMP.10.a executed"
		clr_country_flag = IMP_invite_pending
		IMP_admit_full_member = yes
		unlock_decision_category_tooltip = IMP_council_category
		hidden_effect = { FROM = { country_event = { id = IMP.12 days = 1 } } }
		ai_chance = {
			base = 10
			modifier = { add = 5 has_opinion = { target = FROM value > 25 } }
			modifier = { factor = 0 has_opinion = { target = FROM value < -25 } }
		}
	}

	option = {
		name = IMP.10.b
		log = "[GetDateText]: [This.GetName]: IMP.10.b executed"
		clr_country_flag = IMP_invite_pending
		hidden_effect = { FROM = { country_event = { id = IMP.13 days = 1 } } }
		ai_chance = { base = 2 }
	}
}
