add_namespace = IMP

# Full membership invitation. ROOT is the invitee, FROM the chair
country_event = {
	id = IMP.10
	title = IMP.10.t
	desc = IMP.10.d
	picture = GFX_belarus_intermarium
	is_triggered_only = yes

	option = {
		name = IMP.10.a
		log = "[GetDateText]: [This.GetName]: IMP.10.a executed"
		IMP_admit_full_member = yes
		clr_country_flag = IMP_invite_pending
		unlock_decision_category_tooltip = IMP_council_category
		hidden_effect = {
			if = {
				limit = { has_idea = IMP_full_member }
				FROM = { country_event = { id = IMP.12 days = 1 } }
			}
			else = {
				FROM = { country_event = { id = IMP.14 days = 1 } }
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = 5 has_opinion = { target = FROM value > 25 } }
			modifier = { factor = 0 has_opinion = { target = FROM value < -25 } }
		}
	}

	option = {
		name = IMP.10.b
		log = "[GetDateText]: [This.GetName]: IMP.10.b executed"
		clr_country_flag = IMP_invite_pending
		hidden_effect = { FROM = { country_event = { id = IMP.13 days = 1 } } }
		ai_chance = { base = 2 }
	}
}

# Program vote. ROOT is a full member other than the proposer
country_event = {
	id = IMP.20
	title = IMP.20.t
	desc = IMP.20.d
	picture = GFX_belarus_intermarium
	is_triggered_only = yes

	option = {
		name = IMP.20.a
		log = "[GetDateText]: [This.GetName]: IMP.20.a executed"
		add_to_variable = { global.IMP_votes_for = 1 }
		ai_chance = {
			base = 10
			modifier = { add = 5 has_opinion = { target = event_target:IMP_vote_proposer value > 25 } }
			modifier = { factor = 0.25 has_opinion = { target = event_target:IMP_vote_proposer value < -25 } }
		}
	}

	option = {
		name = IMP.20.b
		log = "[GetDateText]: [This.GetName]: IMP.20.b executed"
		add_to_variable = { global.IMP_votes_against = 1 }
		ai_chance = {
			base = 2
			modifier = { add = 4 NOT = { has_government = democratic } }
		}
	}
}
