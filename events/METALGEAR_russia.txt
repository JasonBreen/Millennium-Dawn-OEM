country_event = {
	id = METALGEAR.34
	title = METALGEAR.34.t
	desc = METALGEAR.34.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	trigger = {
		METALGEAR_scenario_enabled = yes
		tag = SOV
		country_exists = JAP
		JAP = { check_variable = { METALGEAR_request_phase = 1 } }
	}

	option = {
		name = METALGEAR.34.a
		log = "[GetDateText]: [This.GetName]: METALGEAR.34.a executed"
		custom_effect_tooltip = METALGEAR_accept_exchange_tt
		trigger = {
			country_exists = JAP
			NOT = { has_war_with = JAP }
			JAP = { check_variable = { METALGEAR_request_phase = 1 } }
		}
		hidden_effect = {
			if = {
				limit = {
					country_exists = JAP
					NOT = { has_war_with = JAP }
					JAP = { check_variable = { METALGEAR_request_phase = 1 } }
				}
				add_political_power = -10
				add_opinion_modifier = { target = JAP modifier = diplomatic_support }
				JAP = {
					set_variable = { METALGEAR_request_phase = 2 }
					add_opinion_modifier = { target = SOV modifier = diplomatic_support }
					country_event = { id = METALGEAR.23 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
			modifier = { factor = 2 has_opinion = { target = JAP value > 25 } }
			modifier = { factor = 0.25 has_opinion = { target = JAP value < -25 } }
			modifier = { factor = 1.5 check_variable = { METALGEAR_review_policy = 1 } }
		}
	}

	option = {
		name = METALGEAR.34.b
		log = "[GetDateText]: [This.GetName]: METALGEAR.34.b executed"
		custom_effect_tooltip = METALGEAR_decline_exchange_tt
		trigger = {
			country_exists = JAP
			NOT = { has_war_with = JAP }
			JAP = { check_variable = { METALGEAR_request_phase = 1 } }
		}
		hidden_effect = {
			if = {
				limit = { country_exists = JAP JAP = { check_variable = { METALGEAR_request_phase = 1 } } }
				JAP = {
					set_variable = { METALGEAR_request_phase = 3 }
					country_event = { id = METALGEAR.23 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { factor = 2 has_opinion = { target = JAP value < 0 } }
		}
	}

	option = {
		name = METALGEAR.34.c
		log = "[GetDateText]: [This.GetName]: METALGEAR.34.c executed"
		custom_effect_tooltip = METALGEAR_exchange_interrupted_tt
		trigger = {
			OR = {
				NOT = { country_exists = JAP }
				has_war_with = JAP
				JAP = { NOT = { check_variable = { METALGEAR_request_phase = 1 } } }
			}
		}
		hidden_effect = {
			if = {
				limit = { country_exists = JAP JAP = { check_variable = { METALGEAR_request_phase = 1 } } }
				JAP = {
					set_variable = { METALGEAR_request_phase = 4 }
					country_event = { id = METALGEAR.23 days = 1 }
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}
